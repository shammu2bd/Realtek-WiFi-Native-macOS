 428 IOReturn RTW88WCLController::quiescePacketPath()
 429 {
 430     for (;;) {
 431         uint32_t state = __atomic_load_n(&_packetPathState, __ATOMIC_ACQUIRE);
 432         if (state == kPacketPathDisabled)
 433             return kIOReturnSuccess;
 434         if (state == kPacketPathBinding)
 435             return kIOReturnBusy;
 436         if (state == kPacketPathActive) {
 437             uint32_t expected = kPacketPathActive;
 438             if (!__atomic_compare_exchange_n(&_packetPathState, &expected,
 439                                              kPacketPathQuiescing, false,
 440                                              __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE))
 441                 continue;
 442         } else if (state != kPacketPathQuiescing) {
 443             return kIOReturnBusy;
 444         }
 445         break;
 446     }
 447 
 448     if (__atomic_load_n(&_packetPathInFlight, __ATOMIC_ACQUIRE) != 0)
 449         return kIOReturnBusy;
 450 
 451     uint32_t expected = kPacketPathQuiescing;
 452     if (!__atomic_compare_exchange_n(&_packetPathState, &expected,
 453                                      kPacketPathReleasing, false,
 454                                      __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE))
 455         return kIOReturnBusy;
 456 
 457     IO80211SkywalkInterface *interface = _packetPathInterface;
 458     _packetPathInterface = nullptr;
 459     if (interface)
 460         interface->release();
 461     __atomic_store_n(&_packetPathState, kPacketPathDisabled, __ATOMIC_RELEASE);
 462     return kIOReturnSuccess;
 463 }

  42 IOReturn RTW88WCLPCIBackend::open(IOPCIDevice *provider,
  43                                   IOWorkLoop *workLoop,
  44                                   RTW88WCLPCIBackendClient *client)
  45 {
  46     if (!provider || !workLoop)
  47         return kIOReturnBadArgument;
  48     if (_opened || _pciDev || _compatLeaseHeld)
  49         return kIOReturnBusy;
  50 
  51     _pciDev = provider;
  52     _pciDev->retain();
  53     _workLoop = workLoop;
  54     _workLoop->retain();
  55     _client = client;
  56 
  57     _dmaLock = IOSimpleLockAlloc();
  58     _pendingFreeLock = IOSimpleLockAlloc();
  59     if (!_dmaLock || !_pendingFreeLock) {
  60         closeResources();
  61         return kIOReturnNoMemory;
  62     }
  63 
  64     _pciDev->setMemoryEnable(true);
  65     _pciDev->setBusMasterEnable(true);
  66     const UInt16 enabledCommand = _pciDev->configRead16(kIOPCIConfigCommand);
  67     if ((enabledCommand & 0x0006U) != 0x0006U) {
  68         IOLog("rtl88wifi-wcl: PCI memory/bus-master enable failed (command=0x%04x)\n",
  69               enabledCommand);
  70         closeResources();
  71         return kIOReturnNotReady;
  72     }
  73 
  74     /* rtw88's PCI backend maps BAR2 (the Linux driver uses bar_id=2). */

  76     if (!_mmioMap || !_mmioMap->getVirtualAddress() || !_mmioMap->getLength()) {
  77         IOLog("rtl88wifi-wcl: unable to map a usable BAR2\n");
  78         closeResources();
  79         return kIOReturnNoResources;
  80     }
  81     _mmioBase = reinterpret_cast<volatile void *>(_mmioMap->getVirtualAddress());
  82 
  83     _compatPciDev = static_cast<struct pci_dev *>(IOMallocZero(sizeof(*_compatPciDev)));
  84     if (!_compatPciDev) {
  85         closeResources();
  86         return kIOReturnNoMemory;
  87     }
  88     _compatPciDev->vendor = _pciDev->configRead16(0x00);
  89     _compatPciDev->device = _pciDev->configRead16(0x02);
  90     _compatPciDev->kext_dev = this;
  91     _compatPciDev->resource[2] = reinterpret_cast<resource_size_t>(_mmioBase);
  92     _compatPciDev->resource_len[2] = static_cast<resource_size_t>(_mmioMap->getLength());
  93 
  94     /* The compat layer has process-global callbacks, so acquire this lease
  95      * before publishing the function tables or starting its workqueues. */
  96     RTW88HwOps *expectedOwner = nullptr;
  97     if (!__atomic_compare_exchange_n(&g_pci_dev_instance, &expectedOwner,
  98                                      static_cast<RTW88HwOps *>(this), false,
  99                                      __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE)) {
 100         IOLog("rtl88wifi-wcl: RTW88 compat hardware lease is already owned\n");
 101         closeResources();
 102         return kIOReturnBusy;
 103     }
 104     _compatLeaseHeld = true;
 105     rtw88_pci_io_ops = &_pci_io_ops;
 106     rtw88_dma_ops = &_dma_ops;
 107 
 108     const int compatStatus = rtw88_compat_init();
 109     if (compatStatus != 0) {
 110         IOLog("rtl88wifi-wcl: compat runtime initialization failed (%d)\n",
 111               compatStatus);
 112         closeResources();
 113         return kIOReturnNoResources;
 114     }
 115     _compatInitialized = true;
 116 
 117     _interruptSource = IOInterruptEventSource::interruptEventSource(
 118         this,
 119         OSMemberFunctionCast(IOInterruptEventSource::Action, this,
 120                              &RTW88WCLPCIBackend::handleInterrupt),
 121         _pciDev, 0);
 122     if (!_interruptSource ||
 123         _workLoop->addEventSource(_interruptSource) != kIOReturnSuccess) {
 124         IOLog("rtl88wifi-wcl: unable to register PCI interrupt event source\n");
 125         closeResources();
 126         return kIOReturnNoResources;
 127     }
 128     _interruptSource->disable();
 129 
 130     _opened = true;
 131     IOLog("rtl88wifi-wcl: PCI backend acquired %04x:%04x, BAR2=%p length=0x%llx\n",
 132           _compatPciDev->vendor, _compatPciDev->device, (void *)_mmioBase,
 133           static_cast<unsigned long long>(_mmioMap->getLength()));
 134     return kIOReturnSuccess;

 137 IOReturn RTW88WCLPCIBackend::setRadioRunning(bool running)
 138 {
 139     if (!_opened || !_pciDev || !_interruptSource)
 140         return kIOReturnNotReady;
 141     if (_radioRunning == running)
 142         return kIOReturnSuccess;
 143 
 144     if (running) {
 145         _pciDev->setMemoryEnable(true);
 146         _pciDev->setBusMasterEnable(true);
 147         const UInt16 command = _pciDev->configRead16(kIOPCIConfigCommand);
 148         if ((command & 0x0006U) != 0x0006U) {
 149             _interruptSource->disable();
 150             _pciDev->setBusMasterEnable(false);
 151             const UInt16 rollbackCommand = _pciDev->configRead16(kIOPCIConfigCommand);
 152             if (rollbackCommand & 0x0004U) {
 153                 /* Bus mastering may still be active. Keep the backend marked
 154                  * running so close/free cannot reclaim DMA mappings. */
 155                 _radioRunning = true;
 156                 IOLog("rtl88wifi-wcl: partial PCI enable rollback failed (command=0x%04x); DMA resources quarantined\n",
 157                       rollbackCommand);
 158             } else {
 159                 _radioRunning = false;
 160                 _pciDev->setMemoryEnable(false);
 161             }
 162             return kIOReturnNotReady;
 163         }
 164         _radioRunning = true;
 165         _interruptSource->enable();
 166         return kIOReturnSuccess;
 167     }
 168 
 169     /* Caller contract: RTW88IEEE80211::stop() must already have synchronized
 170      * its interrupt/NAPI workers and stopped device DMA before this call. */
 171     _interruptSource->disable();
 172     _pciDev->setBusMasterEnable(false);
 173     const UInt16 command = _pciDev->configRead16(kIOPCIConfigCommand);
 174     if (command & 0x0004U) {
 175         IOLog("rtl88wifi-wcl: PCI bus-master fence did not stick (command=0x%04x)\n",
 176               command);
 177         return kIOReturnNotReady;
 178     }
 179     _radioRunning = false;
 180     if (preemption_enabled())
 181         drainPendingFree();
 182     return kIOReturnSuccess;
 183 }

 193 void RTW88WCLPCIBackend::closeResources()
 194 {
 195     if (_interruptSource) {
 196         _interruptSource->disable();
 197         if (_workLoop)
 198             _workLoop->removeEventSource(_interruptSource);
 199         _interruptSource->release();
 200         _interruptSource = nullptr;
 201     }
 202 
 203     /* Stop all compat work before reclaiming DMA buffers and releasing the
 204      * callbacks they may invoke. The RTW88 core must already be stopped. */
 205     if (_compatInitialized) {
 206         rtw88_compat_exit();
 207         _compatInitialized = false;
 208     }
 209 
 210     if (preemption_enabled())
 211         drainPendingFree();
 212     releaseDMAEntries();
 213 
 214     clearCompatLease();
 215 
 216     if (_pciDev) {
 217         _pciDev->setBusMasterEnable(false);
 218         _pciDev->setMemoryEnable(false);
 219     }
 220     if (_compatPciDev) {
 221         IOFree(_compatPciDev, sizeof(*_compatPciDev));
 222         _compatPciDev = nullptr;
 223     }
 224     if (_mmioMap) {
 225         _mmioMap->release();
 226         _mmioMap = nullptr;
 227     }
 228     _mmioBase = nullptr;
 229 
 230     if (_workLoop) {
 231         _workLoop->release();
 232         _workLoop = nullptr;
 233     }
 234     if (_pciDev) {
 235         _pciDev->release();
 236         _pciDev = nullptr;
 237     }
 238     if (_pendingFreeLock) {
 239         IOSimpleLockFree(_pendingFreeLock);
 240         _pendingFreeLock = nullptr;
 241     }
 242     if (_dmaLock) {
 243         IOSimpleLockFree(_dmaLock);
 244         _dmaLock = nullptr;
 245     }
 246 
 247     _client = nullptr;
 248     _radioRunning = false;
 249     _opened = false;

 330 void *RTW88WCLPCIBackend::allocCoherent(size_t size,
 331                                          IOPhysicalAddress *physical)
 332 {
 333     if (physical) *physical = 0;
 334     if (!size || !_compatLeaseHeld || !_dmaLock || !_pendingFreeLock)
 335         return nullptr;
 336     if (preemption_enabled())
 337         drainPendingFree();

 339     IOBufferMemoryDescriptor *descriptor =
 340         IOBufferMemoryDescriptor::inTaskWithPhysicalMask(
 341             kernel_task,
 342             kIOMemoryPhysicallyContiguous | kIODirectionInOut |
 343                 kIOMemoryKernelUserShared,
 344             size, 0x00000000FFFFFFF0ULL);
 345     if (!descriptor)
 346         return nullptr;
 347     if (descriptor->prepare() != kIOReturnSuccess) {
 348         descriptor->release();
 349         return nullptr;
 350     }

 352     const IOPhysicalAddress address = descriptor->getPhysicalAddress();
 353     void *virtualAddress = descriptor->getBytesNoCopy();
 354     if (!virtualAddress || !address) {
 355         descriptor->complete();
 356         descriptor->release();
 357         return nullptr;
 358     }
 359     bzero(virtualAddress, size);
 360 
 361     RTW88DMAEntry *entry = static_cast<RTW88DMAEntry *>(IOMallocZero(sizeof(*entry)));
 362     if (!entry) {
 363         descriptor->complete();
 364         descriptor->release();
 365         return nullptr;
 366     }
 367     entry->desc = descriptor;
 368     entry->virt = virtualAddress;
 369     entry->phys = address;
 370     entry->size = size;

 373     entry->next = _dmaList;
 374     _dmaList = entry;
 375     IOSimpleLockUnlock(_dmaLock);
 376     if (physical) *physical = address;

 457 void RTW88WCLPCIBackend::syncBounceForCpu(IOPhysicalAddress dmaAddress,
 458                                           size_t size)
 459 {
 460     if (!_dmaLock || !dmaAddress || !size)
 461         return;
 462     /* Keep the entry pinned by the list lock for the bounded copy; this avoids
 463      * racing a concurrent unmap/free against the original buffer pointer. */
 464     IOSimpleLockLock(_dmaLock);
 465     for (RTW88DMAEntry *entry = _dmaList; entry; entry = entry->next) {
 466         if (entry->phys == dmaAddress && entry->orig_va && entry->virt) {
 467             const size_t copyLength = size < entry->size ? size : entry->size;
 468             memcpy(entry->orig_va, entry->virt, copyLength);
 469             break;
 470         }
 471     }
 472     IOSimpleLockUnlock(_dmaLock);
 473 }
