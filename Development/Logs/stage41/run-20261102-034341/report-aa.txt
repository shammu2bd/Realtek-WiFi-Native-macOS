  193 {
  194     const IOReturn packetPathStatus = quiescePacketPath();

  196         IOLog("rtl88wifi-wcl: packet path quiesce deferred: 0x%08x\n",
  197               packetPathStatus);
  198         return packetPathStatus; // Keep the radio, interface and PCI callbacks alive.
  199     }
  200 
  201     /* Stop the RTW88 RX/TX producer before removing its deferred consumers or
  202      * disabling PCI interrupts. The staged path currently leaves
  203      * _radioStarted false; preserve this ordering when native radio start is
  204      * enabled after queue registration. */
  205     if (_radio && _radioStarted) {
  206         _radio->stop();
  207         _radioStarted = false;
  208     }
  209

  211         const IOReturn fenceStatus = _pciBackend->setRadioRunning(false);
  212         if (fenceStatus != kIOReturnSuccess) {

  214                   fenceStatus);
  215             return fenceStatus; // Keep callbacks, mappings and DMA owner alive on failure.
  216         }
  217     }
  218 
  219     /* Remove the RX workloop source only after the RTW88 producer is stopped.
  220      * stop() waits out an active checkForWork() and releases all staged mbufs. */
  221     if (_rxAdapter) {
  222         _rxAdapter->stop();
  223         _rxAdapter->release();
  224         _rxAdapter = nullptr;
  225     }
  226 
  227     /* The RX adapter transfers accepted native packets into the controller's
  228      * bounded ready ring. Recycle those packets while their pool still exists;
  229      * on a pool error, retain the queue owner and PCI/backend context so a later
  230      * stop/free attempt can retry rather than releasing live packet memory. */

  240     if (_skywalkQueues) {

  387 bool RTW88WCLController::enterPacketPath()
  388 {
  389     if (__atomic_load_n(&_packetPathState, __ATOMIC_ACQUIRE) !=
  390         kPacketPathActive)
  391         return false;
  392 
  393     __atomic_add_fetch(&_packetPathInFlight, 1U, __ATOMIC_ACQ_REL);
  394     if (__atomic_load_n(&_packetPathState, __ATOMIC_ACQUIRE) ==
  395         kPacketPathActive)
  396         return true;
  397 
  398     __atomic_sub_fetch(&_packetPathInFlight, 1U, __ATOMIC_ACQ_REL);
  399     return false;
  400 }

  428 IOReturn RTW88WCLController::quiescePacketPath()
  429 {
  430     for (;;) {
  431         uint32_t state = __atomic_load_n(&_packetPathState, __ATOMIC_ACQUIRE);
  432         if (state == kPacketPathDisabled)
  433             return kIOReturnSuccess;
  434         if (state == kPacketPathBinding)
  435             return kIOReturnBusy;
  436         if (state == kPacketPathActive) {
  437             uint32_t expected = kPacketPathActive;
  438             if (!__atomic_compare_exchange_n(&_packetPathState, &expected,
  439                                              kPacketPathQuiescing, false,
  440                                              __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE))
  441                 continue;
  442         } else if (state != kPacketPathQuiescing) {
  443             return kIOReturnBusy;
  444         }
  445         break;
  446     }
  447 
  448     if (__atomic_load_n(&_packetPathInFlight, __ATOMIC_ACQUIRE) != 0)
  449         return kIOReturnBusy;
  450 
  451     uint32_t expected = kPacketPathQuiescing;
  452     if (!__atomic_compare_exchange_n(&_packetPathState, &expected,
  453                                      kPacketPathReleasing, false,
  454                                      __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE))
  455         return kIOReturnBusy;
  456 
  457     IO80211SkywalkInterface *interface = _packetPathInterface;
  458     _packetPathInterface = nullptr;
  459     if (interface)
  460         interface->release();
  461     __atomic_store_n(&_packetPathState, kPacketPathDisabled, __ATOMIC_RELEASE);
  462     return kIOReturnSuccess;
  463 }

  526 IOReturn RTW88WCLController::recycleStagedRxPackets()
  527 {
  528     if (_stagedRxPackets.empty())
  529         return kIOReturnSuccess;
  530     if (!_skywalkQueues || !_skywalkQueues->rxPacketPool())
  531         return kIOReturnNotReady;
  532 
  533     IOSkywalkPacketBufferPool *pool = _skywalkQueues->rxPacketPool();
  534     IOSkywalkPacket *packet = nullptr;
  535     bool disposalRequired = false;
  536     while (_stagedRxPackets.pop(&packet)) {
  537         if (!packet)
  538             continue;
  539 
  540         const IOReturn status = pool->deallocatePacket(packet);
  541         if (status != kIOReturnSuccess) {
  542             /* The radio and RX event source are already quiesced, and these
  543              * queues were never registered. If a packet cannot be returned
  544              * individually, dispose the pool's packet set as one teardown
  545              * operation before its memory segments are released. */
  546             IOLog("rtl88wifi-wcl: RX packet recycle returned 0x%08x; pool disposal required\n",
  547                   status);
  548             disposalRequired = true;
  549         }
  550         packet = nullptr;
  551     }
  552 
  553     if (disposalRequired)
  554         pool->disposeAllPackets();

  560 {
  561     auto *self = static_cast<RTW88WCLController *>(context);
  562     if (!self || !preparedPacket)
  563         return false;

  565     /* Preserve the pool-backed packet in a fixed-capacity owner ring. The
  566      * ring is bounded to the RX pool's 128 packets; overflow returns false,
  567      * causing the RX adapter to recycle the packet immediately. No queue API
  568      * is called here: completion-queue registration/ownership is still fenced. */
  569     return self->_stagedRxPackets.push(preparedPacket);
  570 }

  102 static bool isZeroAddress(const uint8_t *address, size_t length)

  110 static bool validInformationElements(const uint8_t *ies, size_t length)
  111 {
  112     if (length && !ies) return false;
  113     size_t offset = 0;
  114     while (offset < length) {
  115         if (length - offset < 2) return false;
  116         const size_t elementLength = ies[offset + 1];
  117         if (elementLength > length - offset - 2) return false;
  118         offset += 2 + elementLength;
  119     }
  120     return true;
  121 }

  173 IOReturn RTW88WCLInfraInterface::attachRadio(RTW88IEEE80211 *radio)
  174 {
  175     if (!radio || !_controller)
  176         return kIOReturnBadArgument;
  177     if (_bridge.isAttached() || _workQueue)
  178         return kIOReturnBusy;
  179 
  180     IOWorkLoop *workLoop = _controller->getWorkLoop();
  181     if (!workLoop)
  182         return kIOReturnNotReady;
  183 
  184     RTW88WCLWorkQueue *queue = new RTW88WCLWorkQueue;
  185     if (!queue)
  186         return kIOReturnNoMemory;
  187     if (!queue->init(this, workLoop, this)) {
  188         queue->release();
  189         return kIOReturnNoResources;
  190     }
  191 
  192     IOReturn status = _bridge.attach(radio, queue);
  193     if (status != kIOReturnSuccess) {
  194         queue->stop();
  195         queue->release();
  196         return status;
  197     }
  198     _workQueue = queue;
  199     _stopping = false;
  200     return kIOReturnSuccess;
  201 }

  235 bool RTW88WCLInfraInterface::prepareRegistrationInfo(const ether_addr &address)
  236 {
  237     using namespace RTW88WCLRegistrationInfo24H32;
  238 
  239     clearRegistrationInfo();
  240     auto *info = reinterpret_cast<IOSkywalkEthernetInterface::RegistrationInfo *>(
  241         _registrationInfo.bytes);
  242 
  243     /* 24H32 initRegistrationInfo requires option 1 and the exact 0x130 size.
  244      * It supplies the other opaque slots through this interface's virtual

  247     if (!initRegistrationInfo(info, 1, sizeof(_registrationInfo.bytes))) {

  253         !matchesApple24H32Template(_registrationInfo, address.octet)) {
  254         clearRegistrationInfo();
  255         return false;
  256     }
  257

  259     return true;
  260 }

  283 IOReturn RTW88WCLInfraInterface::enable(unsigned int options)
  284 {
  285     (void)options;
  286     if (_stopping || !_bridge.isAttached() || !_controller)
  287         return kIOReturnNotReady;
  288 
  289     /* Bring the RTW88 station path up first. Only then may the controller's
  290      * standard mbuf TX hook admit packets for this registered interface. */
  291     IOReturn status = _bridge.powerOn();
  292     if (status != kIOReturnSuccess)
  293         return status;
  294 
  295     status = _controller->activatePacketPath(this);
  296     if (status != kIOReturnSuccess) {
  297         _bridge.powerOff();
  298         return status;
  299     }
  300     return kIOReturnSuccess;
  301 }

  304 {
  305     (void)options;
  306     if (!_bridge.isAttached() || !_controller)
  307         return kIOReturnNotReady;

  523 IOReturn RTW88WCLInfraInterface::setWCL_SCAN_REQ(apple80211ScanRequest *data)
  524 {
  525     if (!data) return kIOReturnBadArgument;
  526     if (!_bridge.isAttached() || _stopping) return kIOReturnNotReady;
  527     if (_scanActive || _bridge.scanInFlight()) return kIOReturnBusy;
  528 
  529     const uint8_t *raw = reinterpret_cast<const uint8_t *>(data);
  530     const uint32_t channelCount = RTW88WCLRequest::read32(raw + 0x54);
  531     if (channelCount > 400) return kIOReturnBadArgument;
  532     const size_t requestLength = 0x58 + static_cast<size_t>(channelCount) * 12;
  533     RTW88WCLRequest::Scan request = {};
  534     if (!RTW88WCLRequest::decodeScan(data, requestLength, request))
  535         return kIOReturnBadArgument;
  536 
  537     /* The backend currently supports active scans only. A FAST request is
  538      * serviced as a fresh active scan because there is no separate cached-
  539      * scan transaction; passive/background requests fail closed rather than
  540      * being silently converted into active probe transmission. */
  541     if (request.type != APPLE80211_SCAN_TYPE_ACTIVE &&
  542         request.type != APPLE80211_SCAN_TYPE_FAST)
  543         return kIOReturnUnsupported;
  544 
  545     uint16_t channels[128] = {};
  546     uint32_t selected = 0;
  547     for (unsigned channel = 1; channel <= 196; ++channel) {
  548         if (!(request.channelBitmap[channel / 8] & (1U << (channel % 8))))
  549             continue;
  550         if (selected == sizeof(channels) / sizeof(channels[0]))
  551             return kIOReturnNoResources;
  552         channels[selected++] = static_cast<uint16_t>(channel);
  553     }
  554 
  555     if (request.channelCount != 0 && selected == 0)
  556         return kIOReturnUnsupported;
  557 
  558     ScanFilter filter = {};
  559     filter.identifier = request.identifier;
  560     filter.ssidLength = static_cast<uint8_t>(request.ssidLength);
  561     memcpy(filter.ssid, request.ssid, sizeof(filter.ssid));
  562     memcpy(filter.bssid, request.bssid, sizeof(filter.bssid));
  563     memcpy(filter.channelBitmap, request.channelBitmap,
  564            sizeof(filter.channelBitmap));
  565     filter.filterBSSID = !isZeroAddress(filter.bssid, sizeof(filter.bssid));
  566     filter.filterChannels = request.channelCount != 0;
  567     _scanFilter = filter;
  568     _scanActive = true;
  569     /* The bridge writes this generation before it enters the asynchronous
  570      * backend, so a fast completion cannot race the interface's filter state. */
  571     _activeScanGeneration = 0;
  572     setScanningState(request.identifier, true, nullptr, 0);
  573 
  574     IOReturn status = _bridge.startScan(selected ? channels : nullptr, selected,
  575                                         &_activeScanGeneration);
  576     if (status != kIOReturnSuccess) {
  577         _scanActive = false;
  578         _activeScanGeneration = 0;
  579         setScanningState(request.identifier, false, nullptr, 0);
  580     }
  581     return status;
  582 }

  584 IOReturn RTW88WCLInfraInterface::setWCL_ASSOCIATE(
  585     apple80211AssocCandidates *data)
  586 {
  587     if (!data) return kIOReturnBadArgument;
  588     if (!_bridge.isAttached() || _stopping) return kIOReturnNotReady;
  589 
  590     const uint8_t *raw = reinterpret_cast<const uint8_t *>(data);
  591     const uint32_t candidates = RTW88WCLRequest::read32(raw + 0x214);
  592     if (candidates == 0 || candidates > 10) return kIOReturnBadArgument;
  593     const size_t requestLength = 0x218 + static_cast<size_t>(candidates) * 18;
  594     RTW88WCLRequest::Join request = {};
  595     if (!RTW88WCLRequest::decodeJoin(data, requestLength, request))
  596         return kIOReturnUnsupported;
  597 
  598     const uint8_t *pmk = request.keyLength == 32 ? request.key : nullptr;
  599     const uint8_t *rsn = request.rsnLength ? request.rsn : nullptr;
  600     _assocNotified = false;
  601     _connectNotified = false;
  602     memcpy(_targetBSSID, request.bssid, sizeof(_targetBSSID));
  603     /* The bridge writes this generation before cmdConnect() can schedule
  604      * authentication or produce an event. Duplicate pending joins reuse the
  605      * existing generation. */
  606     _activeJoinGeneration = 0;
  607     IOReturn status = _bridge.associate(request.ssid,
  608                                         static_cast<uint8_t>(request.ssidLength),
  609                                         request.bssid, pmk, rsn,
  610                                         request.rsnLength, false /* internal RTW88 supplicant */,
  611                                         &_activeJoinGeneration);
  612     bzero(request.key, sizeof(request.key));
  613     if (status != kIOReturnSuccess) {
  614         _activeJoinGeneration = 0;
  615         bzero(_targetBSSID, sizeof(_targetBSSID));
  616         return status;
  617     }
  618     return kIOReturnSuccess;
  619 }

  628 IOReturn RTW88WCLInfraInterface::setWCL_LINK_STATE_UPDATE(
  629     apple80211_wcl_update_link_state *data)
  630 {
  631     if (!data) return kIOReturnBadArgument;
  632 
  633     /* Sequoia 24H32 WCL wire layout: byte 6 indicates an up-state record,
  634      * byte 8 says the BSSID field is valid, byte 7 indicates link-up, and
  635      * bytes 12..15 carry the down-state debounce interval. These offsets
  636      * match the reviewed 24H32 WCL protocol layout; don't cast the opaque
  637      * private structure to an older apple80211 definition. */
  638     const uint8_t *bytes = reinterpret_cast<const uint8_t *>(data);
  639     if (bytes[6]) {
  640         if (bytes[8])
  641             setCurrentApAddress(reinterpret_cast<ether_addr *>(data));
  642         if (bytes[7])
  643             setLinkState(kIO80211NetworkLinkUp, 0, false, 0);
  644         return kIOReturnSuccess;
  645     }
  646 
  647     const uint32_t debounce = RTW88WCLRequest::read32(bytes + 12);
  648     setCurrentApAddress(nullptr);
  649     setLinkState(kIO80211NetworkLinkDown, 0, debounce != 0, debounce);
  650     return kIOReturnSuccess;
  651 }

  653 bool RTW88WCLInfraInterface::scanResultMatches(const RTW88BSS &bss) const
  654 {
  655     if (bss.channel == 0 || bss.channel > 196 || bss.ssid_len > 32 ||
  656         bss.ies_len > sizeof(bss.ies) ||
  657         !validInformationElements(bss.ies, bss.ies_len)) return false;
  658     if (_scanFilter.filterBSSID &&
  659         memcmp(bss.bssid, _scanFilter.bssid, sizeof(_scanFilter.bssid)) != 0)
  660         return false;
  661     if (_scanFilter.ssidLength &&
  662         (bss.ssid_len != _scanFilter.ssidLength ||
  663          memcmp(bss.ssid, _scanFilter.ssid, _scanFilter.ssidLength) != 0))
  664         return false;
  665     if (_scanFilter.filterChannels &&
  666         !(_scanFilter.channelBitmap[bss.channel / 8] &
  667           (1U << (bss.channel % 8))))
  668         return false;
  669     return true;
  670 }

  680 void RTW88WCLInfraInterface::publishScanResults(IOReturn status)
  681 {
  682     if (!_controller) return;
  683     if (status == kIOReturnSuccess) {
  684         for (uint32_t index = 0; index < 128; ++index) {
  685             RTW88BSS bss = {};
  686             IOReturn copied = _bridge.copyScanBSS(index, &bss);
  687             if (copied == kIOReturnNotFound) break;
  688             if (copied != kIOReturnSuccess) break;
  689             if (!scanResultMatches(bss)) continue;
  690 
  691             uint8_t message[sizeof(RTW88WCLBeaconMetadata) + sizeof(bss.ies)] = {};
  692             RTW88WCLBeaconMetadata metadata = {};
  693             metadata.ieLength = bss.ies_len;
  694             metadata.channelSpec = static_cast<uint16_t>(bss.channel |
  695                 (bss.channel <= 14 ? 0x1000 : 0xd000));
  696             memcpy(metadata.ssid, bss.ssid, bss.ssid_len);
  697             metadata.ssidLength = bss.ssid_len;
  698             metadata.channel = bss.channel;
  699             memcpy(metadata.bssid, bss.bssid, sizeof(metadata.bssid));
  700             metadata.rssi = bss.rssi;
  701             metadata.interval = bss.beacon_interval;
  702             metadata.capability = static_cast<uint16_t>(bss.capabilities);
  703             metadata.flags = kWCLBeaconRSSIValid;
  704             if (bss.ssid_len) metadata.flags |= kWCLBeaconSSIDPresent;
  705             memcpy(message, &metadata, sizeof(metadata));
  706             if (bss.ies_len)
  707                 memcpy(message + sizeof(metadata), bss.ies, bss.ies_len);
  708             postWCLMessage(kWCLMessageScanResult, message,
  709                            sizeof(metadata) + bss.ies_len);
  710         }
  711     }
  712 
  713     _scanActive = false;
  714     _activeScanGeneration = 0;
  715     setScanningState(_scanFilter.identifier, false, nullptr, 0);
  716     postWCLMessage(kWCLMessageScanDone, &status, sizeof(status));
  717 }

  721                                                      IOReturn driverStatus)

  724     event.status = status;

  731 void RTW88WCLInfraInterface::maybePublishConnected()
  732 {
  733     /* setWCL_LINK_UP_DONE may be delivered after a leave/abort request. Never
  734      * publish an old backend CONNECTED state without a live join transaction. */
  735     if (!_activeJoinGeneration || _connectNotified || !_bridge.isAttached() ||
  736         !_bridge.connectionReady())
  737         return;
  738 
  739     _connectNotified = true;
  740     _assocNotified = true;
  741     postConnectionComplete(0, 0, kIOReturnSuccess);
  742     setLinkState(kIO80211NetworkLinkUp, 0, false, 0);
  743 
  744     RTW88WCLLinkStatusEvent link = {};
  745     memcpy(link.bssid, _targetBSSID, sizeof(link.bssid));
  746     link.up = 1;
  747     postWCLMessage(kWCLMessageLinkStatus, &link, sizeof(link));
  748 }

  750 void RTW88WCLInfraInterface::handleRTW88WCLRadioEvent(
  751     const RTW88WCLRadioEvent &event)
  752 {
  753     if (_stopping) return;
  754     switch (event.event) {
  755         case kRTW88EventScanDone: {
  756             if (!_scanActive || event.scanGeneration != _activeScanGeneration)
  757                 return; // stale completion from an aborted/superseded scan
  758             const IOReturn status = event.value ? kIOReturnAborted
  759                                                  : kIOReturnSuccess;
  760             publishScanResults(status);
  761             break;
  762         }
  763         case kRTW88EventConnected: {
  764             if (!_activeJoinGeneration ||
  765                 event.joinGeneration != _activeJoinGeneration)
  766                 return;
  767             if (!_assocNotified) {
  768                 RTW88WCLAssociationEvent assoc = {};
  769                 memcpy(assoc.bssid, _targetBSSID, sizeof(assoc.bssid));
  770                 postWCLMessage(kWCLMessageAssocComplete, &assoc, sizeof(assoc));
  771                 _assocNotified = true;
  772             }
  773             maybePublishConnected();
  774             break;
  775         }
  776         case kRTW88EventAssocDone: {
  777             if (!_activeJoinGeneration ||
  778                 event.joinGeneration != _activeJoinGeneration)
  779                 return; // stale association callback from an earlier join
  780             if (!_assocNotified) {
  781                 RTW88WCLAssociationEvent assoc = {};
  782                 memcpy(assoc.bssid, _targetBSSID, sizeof(assoc.bssid));
  783                 postWCLMessage(kWCLMessageAssocComplete, &assoc, sizeof(assoc));
  784                 _assocNotified = true;
  785             }
  786             maybePublishConnected();
  787             break;
  788         }
  789         case kRTW88EventDeauth: {
  790             if (!_activeJoinGeneration ||
  791                 event.joinGeneration != _activeJoinGeneration)
  792                 return;
  793             uint32_t reason = event.value;
  794             postWCLMessage(kWCLMessageDeauth, &reason, sizeof(reason));
  795             postConnectionComplete(1, static_cast<uint16_t>(reason),
  796                                    kIOReturnNotAttached);
  797             _connectNotified = false;
  798             _assocNotified = false;
  799             _activeJoinGeneration = 0;
  800             setCurrentApAddress(nullptr);
  801             setLinkState(kIO80211NetworkLinkDown, reason, false, 0);
  802             break;
  803         }
  804         case kRTW88EventDisconnected: {
  805             if (!_activeJoinGeneration ||
  806                 event.joinGeneration != _activeJoinGeneration)
  807                 return;
  808             if (_connectNotified || _assocNotified)
  809                 postConnectionComplete(1, 0, kIOReturnNotAttached);
  810             _connectNotified = false;
  811             _assocNotified = false;
  812             _activeJoinGeneration = 0;
  813             setCurrentApAddress(nullptr);
  814             setLinkState(kIO80211NetworkLinkDown, 0, false, 0);
  815             RTW88WCLLinkStatusEvent link = {};
  816             memcpy(link.bssid, _targetBSSID, sizeof(link.bssid));
  817             link.up = 0;
  818             postWCLMessage(kWCLMessageLinkStatus, &link, sizeof(link));
  819             break;
  820         }
  821         case kRTW88EventRSSIChanged: {
  822             if (!_activeJoinGeneration ||
  823                 event.joinGeneration != _activeJoinGeneration)
  824                 return;
  825             int rssi = 0;
  826             if (_bridge.getRSSI(&rssi) == kIOReturnSuccess) {
  827                 RTW88WCLLqmUpdate update = {};
  828                 update.rssiValid = 1;
  829                 update.rssi = rssi;
  830                 postWCLMessage(kWCLMessageLqmUpdate, &update, sizeof(update));
  831             }
  832             break;
  833         }
  834     }
  835 }

  127     CompletionSink sink, void *context)
  128 {

  130         return 0;
  131 
  132     /* A sink may trigger owner teardown; keep this adapter alive until the
  133      * drain loop has stopped touching its ticket pool. */
  134     retain();
  135     uint32_t drained = 0;
  136     for (;;) {
  137         TicketPool::Ticket *ticket = _tickets.claimCompleted();
  138         if (!ticket)
  139             break;

  141         auto *packet = static_cast<IOSkywalkPacket *>(TicketPool::packet(ticket));
  142         const bool transmitted = TicketPool::transmitted(ticket);
  143 
  144         /* The sink runs in deferred context and borrows the retained packet.
  145          * It must finish any Skywalk queue handoff before returning. */
  146         if (packet)
  147             sink(packet, transmitted, context);
  148 
  149         TicketPool::finish(ticket);
  150         if (packet)
  151             packet->release();
  152         release();

  154     }

  144 IOReturn RTW88WCLSkywalkRxAdapter::enqueueFrame(mbuf_t packet)
  145 {
  146     if (!packet)
  147         return kIOReturnBadArgument;
  148 
  149     if (__atomic_load_n(&_stopping, __ATOMIC_ACQUIRE) ||

  159         return kIOReturnNoResources;
  160     }
  161 
  162     uint32_t expected = 0;

  169 bool RTW88WCLSkywalkRxAdapter::checkForWork()
  170 {
  171     if (__atomic_load_n(&_stopping, __ATOMIC_ACQUIRE))
  172         return false;
  173 
  174     uint32_t drained = 0;
  175     mbuf_t source = nullptr;
  176     while (drained < kDrainBudget && _frames.pop(&source)) {
  177         if (source) {
  178             IOSkywalkPacket *packet = nullptr;
  179             if (copyMbufToSkywalkPacket(_rxPool, source, &packet)) {
  180                 __atomic_add_fetch(&_prepared, 1U, __ATOMIC_RELAXED);
  181                 /* A true return means the sink has accepted ownership. The
  182                  * current controller sink returns false because the RX
  183                  * completion queue is still unregistered. */
  184                 const bool accepted = _sink && _sink(packet, _sinkContext);
  185                 if (!accepted) {
  186                     (void)_rxPool->deallocatePacket(packet);
  187                     __atomic_add_fetch(&_dropped, 1U, __ATOMIC_RELAXED);
  188                 }
  189             } else {
  190                 __atomic_add_fetch(&_dropped, 1U, __ATOMIC_RELAXED);
  191             }
  192             mbuf_freem(source);
  193             source = nullptr;
  194         }
  195         ++drained;
  196     }
  197 
  198     if (drained == kDrainBudget)
  199         return true; // bounded workloop occupancy; schedule another pass
  200 
  201     __atomic_store_n(&_pendingSignal, 0U, __ATOMIC_RELEASE);
  202     if (!_frames.empty()) {
  203         uint32_t expected = 0;
  204         if (__atomic_compare_exchange_n(&_pendingSignal, &expected, 1U,
  205                                         false, __ATOMIC_ACQ_REL,
  206                                         __ATOMIC_ACQUIRE))
  207             return true;
  208     }
  209     return false;
  210 }

  227     }

  229     _sinkContext = nullptr;
