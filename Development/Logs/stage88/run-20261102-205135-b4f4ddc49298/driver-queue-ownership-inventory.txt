0035 IOReturn RTW88WCLSkywalkQueues::createQueues()
0036 {
0037     if (_allocated)
0038         return kIOReturnSuccess;
0039     if (_txPool || _rxPool || _txSubmission || _txCompletion ||
0040         _rxSubmission || _rxCompletion)
0041         return kIOReturnBusy;
0042 
0043     RTW88WCLPoolOptionsAccess::PoolOptions txOptions = {};
0044     txOptions.packetCount = kQueueCapacity;
0045     txOptions.bufferCount = kQueueCapacity;
0046     txOptions.bufferSize = kTxBufferSize;
0047     txOptions.maxBuffersPerPacket = 1;
0048     txOptions.memorySegmentSize = kQueueCapacity * kTxBufferSize;
0049     txOptions.poolFlags = 0;
0050     txOptions._reserved = 0;
0051 
0052     RTW88WCLPoolOptionsAccess::PoolOptions rxOptions = txOptions;
0053     rxOptions.bufferSize = kRxBufferSize;
0054     rxOptions.memorySegmentSize = kQueueCapacity * kRxBufferSize;
0055 
0056     _txPool = IOSkywalkPacketBufferPool::withName(
0057         "RTL88WiFi-WCL-TX", this, kIOSkywalkPacketTypeNetwork, &txOptions);
0058     if (!_txPool)
0059         goto no_memory;
0060 
0061     _rxPool = IOSkywalkPacketBufferPool::withName(
0062         "RTL88WiFi-WCL-RX", this, kIOSkywalkPacketTypeNetwork, &rxOptions);
0063     if (!_rxPool)

0085     if (!_rxSubmission)
0086         goto no_memory;
0087

0140 /*
0141  * Sequoia 24H32 callback contract: although the bundled Skywalk action
0142  * typedefs are declared as IOReturn, the audited queue dispatchers
0143  * consume these callback return values as packet counts. Zero means
0144  * that no packets were accepted or completed. Do not return nonzero
0145  * IOReturn error constants from these action callbacks.
0146  */
0147 IOReturn RTW88WCLSkywalkQueues::txSubmissionAction(
0148     OSObject *owner, IOSkywalkTxSubmissionQueue *queue,
0149     const IOSkywalkPacket **packets, UInt32 packetCount, void *refCon)
0150 {
0151     (void)refCon;
0152     auto *self = OSDynamicCast(RTW88WCLSkywalkQueues, owner);
0153     if (!self || queue != self->_txSubmission || !packets || packetCount == 0)
0154         return static_cast<IOReturn>(0);
0155 
0156     /* Validate every descriptor before a future TX path is allowed to copy
0157      * bytes or hand ownership to RTW88. The callback remains fail-closed:
0158      * a validated packet is not yet accepted until its hardware completion
0159      * cookie is connected to the native completion queue. */
0160     for (UInt32 i = 0; i < packetCount; ++i) {
0161         if (!packets[i])
0162             return static_cast<IOReturn>(0);
0163         RTW88WCLPacketView view = {};
0164         if (!RTW88WCLResolveEthernetTxPacket(
0165                 const_cast<IOSkywalkPacket *>(packets[i]), self->_txPool, &view))
0166             return static_cast<IOReturn>(0);
0167     }
0168 
0169     return static_cast<IOReturn>(0);
0170 }
