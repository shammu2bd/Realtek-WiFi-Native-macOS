    96	static u8 get_tx_ampdu_factor(struct ieee80211_sta *sta)
    97	{
    98		u8 exp = sta->deflink.ht_cap.ampdu_factor;
    99	
   100		/* the least ampdu factor is 8K, and the value in the tx desc is the
   101		 * max aggregation num, which represents val * 2 packets can be
   102		 * aggregated in an AMPDU, so here we should use 8/2=4 as the base
   103		 */
   104		return (BIT(2) << exp) - 1;
   105	}

   166	static void rtw_tx_report_enable(struct rtw_dev *rtwdev,

   169		struct rtw_tx_report *tx_report = &rtwdev->tx_report;

   119	static inline struct sk_buff *alloc_skb(u32 size, gfp_t priority)
   120	{
   121	    u32 alloc_size = size + 64; /* extra for head/tail room */
   122	    u8 *data = (u8 *)kmalloc(alloc_size, priority);
   123	    if (!data) return NULL;
   124

   130	    skb->tail = data;
   131	    skb->end  = data + alloc_size;
