  126 uint32_t RTW88WCLSkywalkTxAdapter::drainCompletions(
  127     CompletionSink sink, void *context)
  128 {
  129     if (!sink)
  130         return 0;
  131 
  132     /* A sink may trigger owner teardown; keep this adapter alive until the
  133      * drain loop has stopped touching its ticket pool. */
  134     retain();
  135     uint32_t drained = 0;
  136     for (;;) {
  137         TicketPool::Ticket *ticket = _tickets.claimCompleted();
  138         if (!ticket)
  139             break;
  140 
  141         auto *packet = static_cast<IOSkywalkPacket *>(TicketPool::packet(ticket));
  142         const bool transmitted = TicketPool::transmitted(ticket);
  143 
  144         /* The sink runs in deferred context and borrows the retained packet.
  145          * It must finish any Skywalk queue handoff before returning. */
  146         if (packet)
  147             sink(packet, transmitted, context);
  148 
  149         TicketPool::finish(ticket);
  150         if (packet)
  151             packet->release();
  152         release();
  153         ++drained;
  154     }
