h();
    }

    if (_enabled) disable(_iface);
    drainPendingFree();   /* release any bounce bufs deferred during TX ISR */

    if (_ieee80211)  { _ieee80211->stop(); _ieee80211->release(); _ieee80211 = nullptr; }

    rtw88_compat_exit();

    if (_debugTimer) { _debugTimer->cancelTimeout(); _workLoop->removeEventSource(_debugTimer); _debugTimer->release(); _debugTimer = nullptr; }
    if (_intrSrc)  { _workLoop->removeEventSource(_intrSrc); _intrSrc->release();  _intrSrc = nullptr; }
    if (_cmdGate)  { _workLoop->removeEventSource(_cmdGate); _cmdGate->release();  _cmdGate = nullptr; }
    if (_txQueue)  { _txQueue->release();   _txQueue = nullptr; }
    if (_iface)    { detachInterface(_iface); _iface->release(); _iface = nullptr; }
    if (_workLoop) { _workLoop->release();   _workLoop = nullptr; }
    if (_mmioMap)  { _mmioMap->release();    _mmioMap = nullptr; _mmioBase = nullptr; }
    if (_pciDev)   { _pciDev->release();     _pciDev = nullptr; }

    g_pci_dev_instance = nullptr;
    rtw88_pci_io_ops   = nullptr;
    rtw88_dma_ops      = nullptr;
}

IOOutputQueue *RTW88PCIDevice::createOutputQueue()
{
    /*
     * Without an output queue, IONetworkController delivers outputPacket()
     * straight from the networking stack, which can call it concurrently from
     * multiple threads.  rtw_pci_tx_write_data() computes the TX buffer-
     * descriptor slot from ring->r.wp and fills it BEFORE taking irq_lock
     * (only the wp increment is locked), so two concurrent submissions fill the
     * same slot and skip the next, leaving a zeroed descriptor in the BE ring.
     * The chip then stalls its TX DMA on that zero descriptor (HW rp frozen,
     * FIFO empty) and the shared DMA wedges RX too.
     *
     * An IOGatedOutputQueue runs every outputPacket() under the work-loop gate,
     * serializing submission so the ring fill is single-threaded.
     */
    return IOGatedOutputQueue::withTarget(this, getWorkLoop(), 256);
}

UInt32 RTW88PCIDevice::outputPacket(mbuf_t m, void *param)
{
    drainPendingFree();
    if (!_enabled || !_ieee80211) {
        freePacket(m);
        return kIOReturnOutputDropped;
    }
    /*
     * Backpressure instead of dropping.  When the BE ring is nearly full,
     * rtw_pci_tx_write_data() would return -ENOSPC and rtw_tx() would FREE the
     * skb — silently dropping it.  Under a sustained transfer those dropped
     * frames (TCP ACKs/data) stall the connection.  Returning
     * kIOReturnOutputStall makes IOGatedOutputQueue hold this exact packet and
     * stop dispatching; resumeTxIfStalled() (fired from the IRQ bottom-half
     * after tx_isr frees slots) re-services the queue.  Threshold leaves
     * headroom so rtw_tx never actually hits -ENOSPC.
     */
    if (rtw88_be_tx_avail() < kRTW88TxStallAvail) {
        _txStalled = true;
        return kIOReturnOutputStall;
    }
    return _ieee80211->outputPacket(m);
}

void RTW88PCIDevice::resumeTxIfStalled()
{
    /* Runs on the IRQ bottom-half thread (no rtw88 locks held). Use async
     * service so we never block on the output-queue gate from here. */
    if (_txStalled && rtw88_be_tx_avail() >= kRTW88TxResumeAvail) {
        _txStalled = false;
        if (_txQueue)
            _txQueue->service(IOBasicOutputQueue::kServiceAsync);
    }
}

/*  RX injection (called from RTW88IEEE80211 on frame receive)         */
/* ------------------------------------------------------------------ */

mbuf_t RTW88PCIDevice::allocateInputPacket(uint32_t len)
{
    /* IONetworkController::allocatePacket returns an mbuf set up exactly the
     * way inputPacket() expects: m_len and m_pkthdr.len are both set and
     * consistent across every segment of the chain.  Hand-rolling this with
     * mbuf_allocpacket left m_len inconsistent with pkthdr.len, which the
     * dlil input validator rejects with "Failed mbuf validity check: len -14". */
    return allocatePacket(len);
}

void RTW88PCIDevice::injectRxFrame(mbuf_t m)
{
    drainPendingFree();
    if (!_iface || !_enabled) {
        freePacket(m);
        return;
    }

    /* Last line of defense: never hand the networking stack a malformed
     * packet.  Validate BOTH length fields — the dlil validator panics on
     * m_len (printed as "len"), not just pkthdr.len.  mbuf_len/mbuf_pkthdr_len
     * return size_t, so a negative m_len shows up as a huge value here. */
    size_t plen = mbuf_pkthdr_len(m);
    size_t mlen = mbuf_len(m);
    if (plen < 14 || plen > 4096 || mlen < 14 || mlen > 4096) {
        IOLog("rtw88: injectRxFrame: dropping bogus mbuf (pkthdr.len=%zu m_len=%zu)\n",
              plen, mlen);
        freePacket(m);
        return;
    }

    /* Queue + flush, matching the proven the reference driver submission path.  Submitting
     * via the input queue keeps frame delivery off whatever thread called us. */
    _iface->inputPacket(m, 0, IONetworkInterface::kInputOptionQueuePacket);
    _iface->flushInputQueue();

    IONetworkData *nd = _iface->getNetworkData(kIONetworkStatsKey);
    if (nd) {
        IONetworkStats *stats = (IONetworkStats *)nd->getBuffer();
        if (stats) stats->inputPackets++;
    }
}

/*  DMA coherent allocation                                             */
/* ------------------------------------------------------------------ */

void *RTW88PCIDevice::allocCoherent(size_t size, IOPhysicalAddress *phys)
{
    /*
     * rtw88 TX ring descriptors store DMA addresses in 32-bit fields.
     * Restrict physical allocation to the first 4GB so truncation to
     * cpu_to_le32() in the ring descriptor is lossless.
     * 0x00000000FFFFFFF0 = below 4GB, 16-byte aligned.
     */
    IOBufferMemoryDescriptor *desc = IOBufferMemoryDescriptor::inTaskWithPhysicalMask(
        kernel_task,
        kIOMemoryPhysicallyContiguous | kIODirectionInOut | kIOMemoryKernelUserShared,
        size,
        0x00000000FFFFFFF0ULL);

    if (!desc) { IOLog("rtw88: dma alloc failed, size=%zu\n", size); return nullptr; }
    desc->prepare();

    IOPhysicalAddress pa = desc->getPhysicalAddress();
    void *va = desc->getBytesNoCopy();
    memset(va, 0, size);

    if (phys) *phys = pa;

    /* Allocate the tracking node OUTSIDE the spinlock — IOMallocZero is
     * sleepable, and IOSimpleLock disables preemption.  Calling a sleepable
     * allocator from a preemption-disabled context panics XNU with
     * "blocking while holding a spinlock". */
    DMAEntry *entry = (DMAEntry *)IOMallocZero(sizeof(DMAEntry));
    if (!entry) {
        desc->complete();
        desc->release();
        return nullptr;
    }
    entry->desc = desc;
    entry->virt = va;
    entry->phys = pa;
    entry->size = size;

    IOSimpleLockLock(_dmaLock);
    entry->next = _dmaList;
    _dmaList = entry;
    IOSimpleLockUnlock(_dmaLock);

    return va;
}

void RTW88PCIDevice::freeCoherent(size_t size, void *virt, IOPhysicalAddress phys)
{
    IOSimpleLockLock(_dmaLock);
    DMAEntry **prev = &_dmaList;
    for (DMAEntry *e = _dmaList; e; e = e->next) {
        if (e->virt == virt) {
            *prev = e->next;
            IOSimpleLockUnlock(_dmaLock);
            if (preemption_enabled()) {
                e->desc->complete();
                e->desc->release();
                IOFree(e, sizeof(*e));
            } else {
                IOSimpleLockLock(_pendingFreeLock);
                e->next = _dmaPendingFree;
                _dmaPendingFree = e;
                IOSimpleLockUnlock(_pendingFreeLock);
            }
            return;
        }
        prev = &e->next;
    }
    IOSimpleLockUnlock(_dmaLock);
    IOLog("rtw88: freeCoherent: virt %p not found\n", virt);
}

void RTW88PCIDevice::freeCoherentByPhys(IOPhysicalAddress phys)
{
    IOSimpleLockLock(_dmaLock);
    DMAEntry **prev = &_dmaList;
    for (DMAEntry *e = _dmaList; e; e = e->next) {
        if (e->phys == phys) {
            *prev = e->next;
            IOSimpleLockUnlock(_dmaLock);
            if (preemption_enabled()) {
                e->desc->complete();
                e->desc->release();
                IOFree(e, sizeof(*e));
            } else {
                IOSimpleLockLock(_pendingFreeLock);
                e->next = _dmaPendingFree;
                _dmaPendingFree = e;
                IOSimpleLockUnlock(_pendingFreeLock);
            }
            return;
        }
        prev = &e->next;
    }
    IOSimpleLockUnlock(_dmaLock);
}

void RTW88PCIDevice::drainPendingFree()
{
    if (!_pendingFreeLock) return;
    IOSimpleLockLock(_pendingFreeLock);
    DMAEntry *list      = _dmaPendingFree;
    _dmaPendingFree     = nullptr;
    IOSimpleLockUnlock(_pendingFreeLock);

    for (DMAEntry *e = list; e; ) {
        DMAEntry *next = e->next;
        e->desc->complete();
        e->desc->release();
        IOFree(e, sizeof(*e));
        e = next;
    }
}

void RTW88PCIDevice::setBounceOrigVA(IOPhysicalAddress phys, void *orig_va)
{
    IOSimpleLockLock(_dmaLock);
    for (DMAEntry *e = _dmaList; e; e = e->next) {
        if (e->phys == phys) {
            e->orig_va = orig_va;
            break;
        }
    }
    IOSimpleLockUnlock(_dmaLock);
}

void RTW88PCIDevice::syncBounceForCpu(IOPhysicalAddress dma, size_t size)
{
    /*
     * Called by dma_sync_single_for_cpu(DMA_FROM_DEVICE) after the chip
     * has finished writing received packet data into the bounce buffer.
     * Copy bounce → original skb->data so the driver can parse the packet.
     */
    IOSimpleLockLock(_dmaLock);
    for (DMAEntry *e = _dmaList; e; e = e->next) {
        if (e->phys == dma && e->orig_va && e->virt) {
            size_t copy_len = (size <= e->size) ? size : e->size;
            IOSimpleLockUnlock(_dmaLock);
            memcpy(e->orig_va, e->virt, copy_len);
            return;
        }
    }
    IOSimpleLockUnlock(_dmaLock);
}

IOReturn RTW88PCIDevice::newUserClient(task_t owningTask, void *securityID,
                                        UInt32 type, OSDictionary *properties,
                                        IOUserClient **handler)
{
    RTW88UserClient *client = new RTW88UserClient;
    if (!client) {
        IOLog("rtw88: RTW88UserClient allocation failed\n");
        return kIOReturnNoMemory;
    }

    /* initWithTask — not init() — binds the Mach task port.
     * Without this the kernel port is never "ready for callouts" and
     * IOServiceOpen returns kIOReturnBadArgument before our code runs. */
    if (!client->initWithTask(owningTask, securityID, type, properties)) {
        IOLog("rtw88: RTW88UserClient::initWithTask failed\n");
        client->release();
        return kIOReturnBadArgument;
    }

    if (!client->attach(this)) {
        IOLog("rtw88: RTW88UserClient::attach failed\n");
        client->release();
        return kIOReturnBadArgument;
    }

    if (!client->start(this)) {
        IOLog("rtw88: RTW88UserClient::start failed\n");
        client->detach(this);
        client->release();
        return kIOReturnBadArgument;
    }

    *handler = client;
    return kIOReturnSuccess;
}

/* SPDX-License-Identifier: GPL-2.0 OR BSD-3-Clause
 * Fixed-capacity event ring shared by the private WCL event dispatcher.
 * This header is deliberately independent of IOKit so its queue semantics
 * can be tested on the host toolchain.
 */
#pragma once

#include <stddef.h>

/* Single-owner ring: callers must provide external synchronization when the
 * producer and consumer can run concurrently. No allocation occurs after
 * construction, and enqueue/dequeue are O(1). */
template <typename T, size_t Capacity>
class RTW88WCLRadioEventRing final {
    static_assert(Capacity > 1, "event ring must have at least two slots");

public:
    bool push(const T &value)
    {
        if (_count == Capacity)
            return false;
        _items[_tail] = value;
        _tail = next(_tail);
        ++_count;
        return true;
    }

    bool pop(T *out)
    {
        if (!out || _count == 0)
            return false;
        *out = _items[_head];
        _head = next(_head);
        --_count;
        return true;
    }

    /* Replace the newest slot only when it matches. Restricting coalescing
     * to adjacent telemetry preserves ordering around scan/link transitions. */
    template <typename Predicate>
    bool replaceLast(Predicate predicate, const T &value)
    {
        if (_count == 0)
            return false;
        const size_t index = (_tail + Capacity - 1) % Capacity;
        if (!predicate(_items[index]))
            return false;
        _items[index] = value;
        return true;
    }

    /* Remove a logical (head-relative) slot while preserving FIFO order. */
    bool removeAt(size_t offset)
    {
        if (offset >= _count)
            return false;
        for (size_t i = offset; i + 1 < _count; ++i) {
            const size_t dst = (_head + i) % Capacity;
            const size_t src = (_head + i + 1) % Capacity;
            _items[dst] = _items[src];
        }
        _tail = (_tail + Capacity - 1) % Capacity;
        --_count;
        return true;
    }

    template <typename Predicate>
    bool removeFirst(Predicate predicate)
    {
        for (size_t offset = 0; offset < _count; ++offset) {
            const size_t index = (_head + offset) % Capacity;
            if (predicate(_items[index]))
                return removeAt(offset);
        }
        return false;
    }

    void clear()
    {
        _head = 0;
        _tail = 0;
        _count = 0;
    }

    size_t size() const { return _count; }
    bool empty() const { return _count == 0; }
    bool full() const { return _count == Capacity; }

private:
    static size_t next(size_t index) { return (index + 1) % Capacity; }

    T _items[Capacity] = {};
    size_t _head = 0;
    size_t _tail = 0;
    size_t _count = 0;
};

bool RTW88WCLWorkQueue::init(OSObject *owner, IOWorkLoop *workLoop,
                             RTW88WCLRadioEventHandler *handler)
{
    if (!owner || !workLoop || !handler || _lock || _workLoop)
        return false;
    if (!super::init(owner, nullptr))
        return false;

    _lock = IOSimpleLockAlloc();
    if (!_lock)
        return false;

    _workLoop = workLoop;
    _handler = handler;
    _events.clear();
    _pendingSignal = 0;
    _stopping = 0;
    _registered = 0;
    _dropped = 0;
    _coalescedRSSI = 0;

    const IOReturn status = workLoop->addEventSource(this);
    if (status != kIOReturnSuccess) {
        _workLoop = nullptr;
        _handler = nullptr;
        IOSimpleLockFree(_lock);
        _lock = nullptr;
        return false;
    }
    _registered = 1;
    return true;
}

void RTW88WCLWorkQueue::stop()
{
    IOSimpleLock *lock = _lock;
    if (!lock)
        return;

    IOInterruptState interruptState = IOSimpleLockLockDisableInterrupt(lock);
    _stopping = 1;
    _handler = nullptr;
    _events.clear();
    _pendingSignal = 0;
    IOSimpleLockUnlockEnableInterrupt(lock, interruptState);

    /* Caller must have detached/quiesced the RTW88 event producer. Removal is
     * synchronous with the IOWorkLoop and waits out an active checkForWork(). */
    if (_registered && _workLoop) {
        _workLoop->removeEventSource(this);
        _registered = 0;
    }
    _workLoop = nullptr;
}

{
    stop();
    if (_lock) {
        IOSimpleLockFree(_lock);
        _lock = nullptr;
    }
    super::free();
}

bool RTW88WCLWorkQueue::isTelemetryEvent(const RTW88WCLRadioEvent &event)
{
    return event.event == kRTW88EventRSSIChanged;
}

bool RTW88WCLWorkQueue::enqueueLocked(const RTW88WCLRadioEvent &event)
{
    if (isTelemetryEvent(event)) {
        const bool replaced = _events.replaceLast(
            [&event](const RTW88WCLRadioEvent &queued) {
                return queued.event == kRTW88EventRSSIChanged &&
                       queued.joinGeneration == event.joinGeneration;
            }, event);
        if (replaced) {
            ++_coalescedRSSI;
            return true;
        }
    }

    if (_events.full() && !isTelemetryEvent(event)) {
        /* RSSI samples are advisory; preserve scan/association/disconnect
         * transitions by evicting the oldest RSSI sample first. */
        const bool evicted = _events.removeFirst(
            [](const RTW88WCLRadioEvent &queued) {
                return queued.event == kRTW88EventRSSIChanged;
            });
        if (evicted)
            ++_dropped;
    }

    if (!_events.push(event)) {
        ++_dropped;
        return false;
    }
    return true;
}

void RTW88WCLWorkQueue::onRTW88WCLRadioEvent(
    const RTW88WCLRadioEvent &event)
{
    IOSimpleLock *lock = _lock;
    if (!lock)
        return;

    bool signal = false;
    IOInterruptState interruptState = IOSimpleLockLockDisableInterrupt(lock);
    if (!_stopping && enqueueLocked(event) && !_pendingSignal) {
        _pendingSignal = 1;
        signal = true;
    }
    IOSimpleLockUnlockEnableInterrupt(lock, interruptState);

    /* IOEventSource::signalWorkAvailable wakes its IOWorkLoop. The event
     * source does the actual callback only in checkForWork() on that loop. */
    if (signal)
        signalWorkAvailable();
}

bool RTW88WCLWorkQueue::checkForWork()
{
    /* Bound one pass so sustained telemetry or state churn cannot monopolize
     * the shared workloop. Returning true asks IOWorkLoop to call us again. */
    uint32_t budget = 32;
    while (budget-- != 0) {
        RTW88WCLRadioEvent event = {};
        RTW88WCLRadioEventHandler *handler = nullptr;
        IOSimpleLock *lock = _lock;
        if (!lock)
            return false;

        IOInterruptState interruptState = IOSimpleLockLockDisableInterrupt(lock);
        if (_stopping || !_events.pop(&event)) {
            if (_stopping)
                _events.clear();
            _pendingSignal = 0;
            IOSimpleLockUnlockEnableInterrupt(lock, interruptState);
            return false;
        }
        handler = _handler;
        IOSimpleLockUnlockEnableInterrupt(lock, interruptState);

        /* WCL controller/interface message translation belongs in this
         * handler, which is guaranteed to execute on the controller workloop. */
        if (handler)
            handler->handleRTW88WCLRadioEvent(event);
    }

    IOSimpleLock *lock = _lock;
    if (!lock)
        return false;
    IOInterruptState interruptState = IOSimpleLockLockDisableInterrupt(lock);
    const bool moreWork = !_stopping && !_events.empty();
    if (!moreWork)
        _pendingSignal = 0;
    IOSimpleLockUnlockEnableInterrupt(lock, interruptState);
    return moreWork;
}

uint32_t RTW88WCLWorkQueue::droppedEvents() const
{
    IOSimpleLock *lock = _lock;
    if (!lock)
        return _dropped;
    IOInterruptState state = IOSimpleLockLockDisableInterrupt(lock);
    const uint32_t value = _dropped;
    IOSimpleLockUnlockEnableInterrupt(lock, state);
    return value;
}

uint32_t RTW88WCLWorkQueue::coalescedRSSIEvents() const
{
    IOSimpleLock *lock = _lock;
    if (!lock)
        return _coalescedRSSI;
    IOInterruptState state = IOSimpleLockLockDisableInterrupt(lock);
    const uint32_t value = _coalescedRSSI;
    IOSimpleLockUnlockEnableInterrupt(lock, state);
    return value;
}

#include "RTW88WCLController.hpp"

#define super IO80211Controller
OSDefineMetaClassAndStructors(RTW88WCLController, IO80211Controller)

bool RTW88WCLController::init(OSDictionary *properties)
{
    if (!super::init(properties))
        return false;
    _workLoop = nullptr;
    _packetPathInterface = nullptr;
    _packetPathState = kPacketPathDisabled;
    _packetPathInFlight = 0;
    _pciBackend = nullptr;
    _skywalkQueues = nullptr;
    _rxAdapter = nullptr;
    _stagedRxPackets.reset();
    _radio = nullptr;
    _baseStarted = false;
    _radioStarted = false;
    return true;
}

bool RTW88WCLController::createWorkLoop()
{
    if (_workLoop)
        return true;
    _workLoop = IOWorkLoop::workLoop();
    return _workLoop != nullptr;
}

IOWorkLoop *RTW88WCLController::getWorkLoop() const
{
    return _workLoop;
}

bool RTW88WCLController::start(IOService *provider)
{
    if (!provider || _baseStarted || _pciBackend || _radio)
        return false;
    if (!super::start(provider))
        return false;
    _baseStarted = true;

    /* Stage A: the controller workloop must exist before acquiring a PCI
     * backend or registering any interrupt source. */
    if (!createWorkLoop()) {
        IOLog("rtl88wifi-wcl: controller workloop creation failed\n");
        goto fail;
    }

    /* Stage B: this native personality must be matched directly to a PCI
     * provider. The backend retains it and owns BAR2/MMIO/DMA resources. */
    {
        IOPCIDevice *pci = OSDynamicCast(IOPCIDevice, provider);
        if (!pci) {
            IOLog("rtl88wifi-wcl: provider is not an IOPCIDevice\n");
            goto fail;
        }

        _pciBackend = new RTW88WCLPCIBackend;
        if (!_pciBackend) {
            IOLog("rtl88wifi-wcl: PCI backend allocation failed\n");
            goto fail;
        }
        if (!_pciBackend->init()) {
            _pciBackend->release();
            _pciBackend = nullptr;
            IOLog("rtl88wifi-wcl: PCI backend initialization failed\n");
            goto fail;
        }

        const IOReturn openStatus = _pciBackend->open(pci, _workLoop, this);
        if (openStatus != kIOReturnSuccess) {
            IOLog("rtl88wifi-wcl: PCI backend open failed: 0x%08x\n",
                  openStatus);
            goto fail;
        }

        /* Stage C: instantiate the RTW88 state machine against the backend's
         * compat pci_dev. This installs callback context and allocates its
         * private timers/workloop only; it does not probe or power the chip. */
        struct pci_dev *compatPCI = _pciBackend->compatPCIDevice();
        if (!compatPCI) {
            IOLog("rtl88wifi-wcl: backend did not provide a compat PCI device\n");
            goto fail;
        }
        RTW88IEEE80211 *radio = RTW88IEEE80211::createAirport(this, compatPCI);
        if (!radio) {
            IOLog("rtl88wifi-wcl: RTW88 state-machine allocation failed\n");
            goto fail;
        }
        radio->setEventDelegate(this);
        const IOReturn bindStatus = bindRadioBackend(radio);
        radio->release();
        if (bindStatus != kIOReturnSuccess) {
            IOLog("rtl88wifi-wcl: RTW88 state-machine bind failed: 0x%08x\n",
                  bindStatus);
            goto fail;
        }
    }

    /* Stage D: reserve native Skywalk queue/pool objects while the PCI and
     * RTW88 owners are still available for orderly unwind. These objects are
     * intentionally not registered or scheduled until packet descriptor
     * conversion and hardware completion correlation are implemented. */
    _skywalkQueues = new RTW88WCLSkywalkQueues;
    if (!_skywalkQueues || !_skywalkQueues->init()) {
        if (_skywalkQueues) {
            _skywalkQueues->release();
            _skywalkQueues = nullptr;
        }
        IOLog("rtl88wifi-wcl: Skywalk queue owner allocation failed\n");
        goto fail;
    }
    {
        const IOReturn queueStatus = _skywalkQueues->createQueues();
        if (queueStatus != kIOReturnSuccess) {
            IOLog("rtl88wifi-wcl: Skywalk queue/pool allocation failed: 0x%08x\n",
                  queueStatus);
            goto fail;
        }
    }

    /* Stage E: bind the RX workloop consumer to the now-owned 4 KiB RX pool.
     * It copies frames into native packet buffers, but its sink remains
     * rejecting until the registration/completion ABI is closed. */
    _rxAdapter = new RTW88WCLSkywalkRxAdapter;
    if (!_rxAdapter || !_rxAdapter->init(
            this, _workLoop, _skywalkQueues->rxPacketPool(),
            &RTW88WCLController::stagedRxFrame, this)) {
        if (_rxAdapter) {
            _rxAdapter->release();
            _rxAdapter = nullptr;
        }
        IOLog("rtl88wifi-wcl: RX packet adapter initialization failed\n");
        goto fail;
    }

    provider->setProperty("RTW88WCLControllerStage",
                           "PCI, RTW88 and pool-backed RX packet conversion staged; datapath pending");

    /* Publication fence: do not probe or power the radio until Skywalk packet
     * registration, RX/TX rings, packet metadata and teardown are implemented.
     * Starting it sooner would produce frames without a safe delivery path. */
    IOLog("rtl88wifi-wcl: backend lifecycle staged; refusing publication until Skywalk datapath is ready\n");
    goto fail;

fail:
    {
        const IOReturn cleanupStatus = releaseNativeResources();
        if (cleanupStatus != kIOReturnSuccess) {
            IOLog("rtl88wifi-wcl: start-failure cleanup incomplete (0x%08x); retaining native resources\n",
                  cleanupStatus);
            return false;
        }
    }
    if (_baseStarted) {
        super::stop(provider);
        _baseStarted = false;
    }
    return false;
}

void RTW88WCLController::stop(IOService *provider)
{
    const IOReturn cleanupStatus = releaseNativeResources();
    if (cleanupStatus != kIOReturnSuccess) {
        IOLog("rtl88wifi-wcl: stop cleanup incomplete (0x%08x); deferring superclass stop and retaining resources\n",
              cleanupStatus);
        return;
    }
    if (_baseStarted) {
        super::stop(provider);
        _baseStarted = false;
    }
}

IOReturn RTW88WCLController::releaseNativeResources()
{
    const IOReturn packetPathStatus = quiescePacketPath();
    if (packetPathStatus != kIOReturnSuccess) {
        IOLog("rtl88wifi-wcl: packet path quiesce deferred: 0x%08x\n",
              packetPathStatus);
        return packetPathStatus; // Keep the radio, interface and PCI callbacks alive.
    }

    /* Stop the RTW88 RX/TX producer before removing its deferred consumers or
     * disabling PCI interrupts. The staged path currently leaves
     * _radioStarted false; preserve this ordering when native radio start is
     * enabled after queue registration. */
    if (_radio && _radioStarted) {
        _radio->stop();
        _radioStarted = false;
    }

    if (_pciBackend && _pciBackend->isRadioRunning()) {
        const IOReturn fenceStatus = _pciBackend->setRadioRunning(false);
        if (fenceStatus != kIOReturnSuccess) {
            IOLog("rtl88wifi-wcl: PCI interrupt/bus-master fence failed: 0x%08x\n",
                  fenceStatus);
            return fenceStatus; // Keep callbacks, mappings and DMA owner alive on failure.
        }
    }

    /* Remove the RX workloop source only after the RTW88 producer is stopped.
     * stop() waits out an active checkForWork() and releases all staged mbufs. */
    if (_rxAdapter) {
        _rxAdapter->stop();
        _rxAdapter->release();
        _rxAdapter = nullptr;
    }

    /* The RX adapter transfers accepted native packets into the controller's
     * bounded ready ring. Recycle those packets while their pool still exists;
     * on a pool error, retain the queue owner and PCI/backend context so a later
     * stop/free attempt can retry rather than releasing live packet memory. */
    const IOReturn recycleStatus = recycleStagedRxPackets();
    if (recycleStatus != kIOReturnSuccess) {
        IOLog("rtl88wifi-wcl: staged RX packet recycle deferred: 0x%08x\n",
              recycleStatus);
        return recycleStatus;
    }

    /* Skywalk packet pools/queues are released only after both the mbuf staging
     * adapter and the prepared-packet ownership ring have been drained. */
    if (_skywalkQueues) {
        _skywalkQueues->destroyQueues();
        _skywalkQueues->release();
        _skywalkQueues = nullptr;
    }

    

    /* RTW88IEEE80211 owns the compat callback context; destroy it before
     * rtw88_compat_exit() and before relinquishing the global hardware lease. */
    unbindRadioBackend();

    if (_pciBackend) {
        const IOReturn closeStatus = _pciBackend->close();
        if (closeStatus != kIOReturnSuccess) {
            IOLog("rtl88wifi-wcl: PCI backend close deferred: 0x%08x\n",
                  closeStatus);
            return closeStatus;
        }
        _pciBackend->release();
        _pciBackend = nullptr;
    }
    return kIOReturnSuccess;
}

void RTW88WCLController::free()
{
    const IOReturn cleanupStatus = releaseNativeResources();
    if (cleanupStatus != kIOReturnSuccess) {
        IOLog("rtl88wifi-wcl: controller shutdown incomplete (0x%08x); quarantining controller and retaining workloop/resources\n",
              cleanupStatus);
        /* Do not release the workloop or call super::free() while any child,
         * packet pool, or PCI/DMA resource may still reference this controller.
         * This deliberately leaks the object rather than risking a UAF/DMA UAF. */
        return;
    }
    if (_workLoop) {
        _workLoop->release();
        _workLoop = nullptr;
    }
    super::free();
}

bool RTW88WCLController::enterPacketPath()
{
    if (__atomic_load_n(&_packetPathState, __ATOMIC_ACQUIRE) !=
        kPacketPathActive)
        return false;

    __atomic_add_fetch(&_packetPathInFlight, 1U, __ATOMIC_ACQ_REL);
    if (__atomic_load_n(&_packetPathState, __ATOMIC_ACQUIRE) ==
        kPacketPathActive)
        return true;

    __atomic_sub_fetch(&_packetPathInFlight, 1U, __ATOMIC_ACQ_REL);
    return false;
}

void RTW88WCLController::leavePacketPath()
{
    __atomic_sub_fetch(&_packetPathInFlight, 1U, __ATOMIC_ACQ_REL);
}

IOReturn RTW88WCLController::activatePacketPath(
    IO80211SkywalkInterface *interface)
{
    if (!interface)
        return kIOReturnBadArgument;
    if (!_radio || !_radioStarted || !_pciBackend || !_pciBackend->isOpen())
        return kIOReturnNotReady;

    uint32_t expected = kPacketPathDisabled;
    if (!__atomic_compare_exchange_n(&_packetPathState, &expected,
                                     kPacketPathBinding, false,
                                     __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE))
        return kIOReturnBusy;

    interface->retain();
    _packetPathInterface = interface;
    __atomic_store_n(&_packetPathState, kPacketPathActive, __ATOMIC_RELEASE);
    IOLog("rtl88wifi-wcl: native mbuf packet handoff enabled\n");
    return kIOReturnSuccess;
}

IOReturn RTW88WCLController::quiescePacketPath()
{
    for (;;) {
        uint32_t state = __atomic_load_n(&_packetPathState, __ATOMIC_ACQUIRE);
        if (state == kPacketPathDisabled)
            return kIOReturnSuccess;
        if (state == kPacketPathBinding)
            return kIOReturnBusy;
        if (state == kPacketPathActive) {
            uint32_t expected = kPacketPathActive;
            if (!__atomic_compare_exchange_n(&_packetPathState, &expected,
                                             kPacketPathQuiescing, false,
                                             __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE))
                continue;
        } else if (state != kPacketPathQuiescing) {
            return kIOReturnBusy;
        }
        break;
    }

    if (__atomic_load_n(&_packetPathInFlight, __ATOMIC_ACQUIRE) != 0)
        return kIOReturnBusy;

    uint32_t expected = kPacketPathQuiescing;
    if (!__atomic_compare_exchange_n(&_packetPathState, &expected,
                                     kPacketPathReleasing, false,
                                     __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE))
        return kIOReturnBusy;

    IO80211SkywalkInterface *interface = _packetPathInterface;
    _packetPathInterface = nullptr;
    if (interface)
        interface->release();
    __atomic_store_n(&_packetPathState, kPacketPathDisabled, __ATOMIC_RELEASE);
    return kIOReturnSuccess;
}

UInt32 RTW88WCLController::outputPacket(mbuf_t packet, void *param)
{
    (void)param;
    if (!packet)
        return kIOReturnOutputDropped;

    if (!enterPacketPath()) {
        mbuf_freem(packet);
        return kIOReturnOutputDropped;
    }

    /* RTW88IEEE80211::outputPacket consumes the mbuf on every return path. */
    const UInt32 result = _radio ? _radio->outputPacket(packet)
                                 : kIOReturnOutputDropped;
    leavePacketPath();
    return result;
}

mbuf_t RTW88WCLController::allocateInputPacket(uint32_t length)
{
    if (length < 14 || length > RTW88WCLSkywalkRxAdapter::kMaximumFrameLength ||
        !_radio || !_rxAdapter || !enterPacketPath())
        return nullptr;

    /* IO80211Controller inherits the IONetworkController packet allocator.
     * Keep its mbuf metadata initialization rather than hand-rolling m_len. */
    mbuf_t packet = allocatePacket(length);
    leavePacketPath();
    return packet;
}

void RTW88WCLController::injectRxFrame(mbuf_t packet)
{
    if (!packet)
        return;

    if (!_rxAdapter || !enterPacketPath()) {
        mbuf_freem(packet);
        return;
    }

    /* enqueueFrame consumes the mbuf on both success and failure. The native
     * RX completion queue is not registered yet, so the workloop sink below
     * deliberately drops frames after staging rather than exposing them. */
    (void)_rxAdapter->enqueueFrame(packet);
    leavePacketPath();
}

IOReturn RTW88WCLController::recycleStagedRxPackets()
{
    if (_stagedRxPackets.empty())
        return kIOReturnSuccess;
    if (!_skywalkQueues || !_skywalkQueues->rxPacketPool())
        return kIOReturnNotReady;

    IOSkywalkPacketBufferPool *pool = _skywalkQueues->rxPacketPool();
    IOSkywalkPacket *packet = nullptr;
    bool disposalRequired = false;
    while (_stagedRxPackets.pop(&packet)) {
        if (!packet)
            continue;

        const IOReturn status = pool->deallocatePacket(packet);
        if (status != kIOReturnSuccess) {
            /* The radio and RX event source are already quiesced, and these
             * queues were never registered. If a packet cannot be returned
             * individually, dispose the pool's packet set as one teardown
             * operation before its memory segments are released. */
            IOLog("rtl88wifi-wcl: RX packet recycle returned 0x%08x; pool disposal required\n",
                  status);
            disposalRequired = true;
        }
        packet = nullptr;
    }

    if (disposalRequired)
        pool->disposeAllPackets();
    return kIOReturnSuccess;
}

bool RTW88WCLController::stagedRxFrame(
    IOSkywalkPacket *preparedPacket, void *context)
{
    auto *self = static_cast<RTW88WCLController *>(context);
    if (!self || !preparedPacket)
        return false;

    /* Preserve the pool-backed packet in a fixed-capacity owner ring. The
     * ring is bounded to the RX pool's 128 packets; overflow returns false,
     * causing the RX adapter to recycle the packet immediately. No queue API
     * is called here: completion-queue registration/ownership is still fenced. */
    return self->_stagedRxPackets.push(preparedPacket);
}

static bool isZeroAddress(const uint8_t *address, size_t length)
{
    if (!address) return true;
    uint8_t aggregate = 0;
    for (size_t i = 0; i < length; ++i) aggregate |= address[i];
    return aggregate == 0;
}

static bool validInformationElements(const uint8_t *ies, size_t length)
{
    if (length && !ies) return false;
    size_t offset = 0;
    while (offset < length) {
        if (length - offset < 2) return false;
        const size_t elementLength = ies[offset + 1];
        if (elementLength > length - offset - 2) return false;
        offset += 2 + elementLength;
    }
    return true;
}

IOReturn RTW88WCLInfraInterface::attachRadio(RTW88IEEE80211 *radio)
{
    if (!radio || !_controller)
        return kIOReturnBadArgument;
    if (_bridge.isAttached() || _workQueue)
        return kIOReturnBusy;

    IOWorkLoop *workLoop = _controller->getWorkLoop();
    if (!workLoop)
        return kIOReturnNotReady;

    RTW88WCLWorkQueue *queue = new RTW88WCLWorkQueue;
    if (!queue)
        return kIOReturnNoMemory;
    if (!queue->init(this, workLoop, this)) {
        queue->release();
        return kIOReturnNoResources;
    }

    IOReturn status = _bridge.attach(radio, queue);
    if (status != kIOReturnSuccess) {
        queue->stop();
        queue->release();
        return status;
    }
    _workQueue = queue;
    _stopping = false;
    return kIOReturnSuccess;
}

void RTW88WCLInfraInterface::detachRadio()
{
    _stopping = true;
    /* Quiesce the producer first, then remove the work source. The controller
     * must stop the radio before destroying its backend object. */
    _bridge.detach();
    if (_workQueue) {
        _workQueue->stop();
        _workQueue->release();
        _workQueue = nullptr;
    }
    _scanActive = false;
    _activeScanGeneration = 0;
    _activeJoinGeneration = 0;
    _assocNotified = false;
    _connectNotified = false;
}

void RTW88WCLInfraInterface::free()
{
    clearRegistrationInfo();
    detachRadio();
    _controller = nullptr;
    super::free();
}

void RTW88WCLInfraInterface::clearRegistrationInfo()
{
    RTW88WCLRegistrationInfo24H32::clear(_registrationInfo);
    _registrationInfoPrepared = false;
}

bool RTW88WCLInfraInterface::prepareRegistrationInfo(const ether_addr &address)
{
    using namespace RTW88WCLRegistrationInfo24H32;

    clearRegistrationInfo();
    auto *info = reinterpret_cast<IOSkywalkEthernetInterface::RegistrationInfo *>(
        _registrationInfo.bytes);

    /* 24H32 initRegistrationInfo requires option 1 and the exact 0x130 size.
     * It supplies the other opaque slots through this interface's virtual
     * hooks; do not replace it with a guessed local structure initializer.
     */
    if (!initRegistrationInfo(info, 1, sizeof(_registrationInfo.bytes))) {
        clearRegistrationInfo();
        return false;
    }

    if (!applyAppleProducerWrites(_registrationInfo, address.octet) ||
        !matchesApple24H32Template(_registrationInfo, address.octet)) {
        clearRegistrationInfo();
        return false;
    }

    _registrationInfoPrepared = true;
    return true;
}

IOReturn RTW88WCLInfraInterface::enable(unsigned int options)
{
    (void)options;
    if (_stopping || !_bridge.isAttached() || !_controller)
        return kIOReturnNotReady;

    /* Bring the RTW88 station path up first. Only then may the controller's
     * standard mbuf TX hook admit packets for this registered interface. */
    IOReturn status = _bridge.powerOn();
    if (status != kIOReturnSuccess)
        return status;

    status = _controller->activatePacketPath(this);
    if (status != kIOReturnSuccess) {
        _bridge.powerOff();
        return status;
    }
    return kIOReturnSuccess;
}

IOReturn RTW88WCLInfraInterface::disable(unsigned int options)
{
    (void)options;
    if (!_bridge.isAttached() || !_controller)
        return kIOReturnNotReady;

    /* Stop new mbuf submissions and wait for synchronous TX handoffs before
     * asking RTW88 to power down. A busy path leaves the radio running. */
    IOReturn status = _controller->quiescePacketPath();
    if (status != kIOReturnSuccess)
        return status;
    return _bridge.powerOff();
}

IOReturn RTW88WCLInfraInterface::getSUPPORTED_CHANNELS(
    apple80211_sup_channel_data *data)

    RTW88Channel channels[APPLE80211_MAX_CHANNELS] = {};
    uint32_t count = 0;
    IOReturn status = _bridge.copyChannels(channels,
                                           APPLE80211_MAX_CHANNELS, &count);
    if (status != kIOReturnSuccess) return status;
    if (count > APPLE80211_MAX_CHANNELS) return kIOReturnNoResources;

    bzero(data, sizeof(*data));
    data->version = APPLE80211_VERSION;
    data->num_channels = count;
    for (uint32_t i = 0; i < count; ++i) {
        if (!channels[i].number || channels[i].number > 196)
            return kIOReturnError;
        data->supported_channels[i].version = APPLE80211_VERSION;
        data->supported_channels[i].channel = channels[i].number;
        data->supported_channels[i].flags = channelFlags(channels[i]);
    }
    return kIOReturnSuccess;
}

IOReturn RTW88WCLInfraInterface::setWCL_SCAN_REQ(apple80211ScanRequest *data)
{
    if (!data) return kIOReturnBadArgument;
    if (!_bridge.isAttached() || _stopping) return kIOReturnNotReady;
    if (_scanActive || _bridge.scanInFlight()) return kIOReturnBusy;

    const uint8_t *raw = reinterpret_cast<const uint8_t *>(data);
    const uint32_t channelCount = RTW88WCLRequest::read32(raw + 0x54);
    if (channelCount > 400) return kIOReturnBadArgument;
    const size_t requestLength = 0x58 + static_cast<size_t>(channelCount) * 12;
    RTW88WCLRequest::Scan request = {};
    if (!RTW88WCLRequest::decodeScan(data, requestLength, request))
        return kIOReturnBadArgument;

    /* The backend currently supports active scans only. A FAST request is
     * serviced as a fresh active scan because there is no separate cached-
     * scan transaction; passive/background requests fail closed rather than
     * being silently converted into active probe transmission. */
    if (request.type != APPLE80211_SCAN_TYPE_ACTIVE &&
        request.type != APPLE80211_SCAN_TYPE_FAST)
        return kIOReturnUnsupported;

    uint16_t channels[128] = {};
    uint32_t selected = 0;
    for (unsigned channel = 1; channel <= 196; ++channel) {
        if (!(request.channelBitmap[channel / 8] & (1U << (channel % 8))))
            continue;
        if (selected == sizeof(channels) / sizeof(channels[0]))
            return kIOReturnNoResources;
        channels[selected++] = static_cast<uint16_t>(channel);
    }

    if (request.channelCount != 0 && selected == 0)
        return kIOReturnUnsupported;

    ScanFilter filter = {};
    filter.identifier = request.identifier;
    filter.ssidLength = static_cast<uint8_t>(request.ssidLength);
    memcpy(filter.ssid, request.ssid, sizeof(filter.ssid));
    memcpy(filter.bssid, request.bssid, sizeof(filter.bssid));
    memcpy(filter.channelBitmap, request.channelBitmap,
           sizeof(filter.channelBitmap));
    filter.filterBSSID = !isZeroAddress(filter.bssid, sizeof(filter.bssid));
    filter.filterChannels = request.channelCount != 0;
    _scanFilter = filter;
    _scanActive = true;
    /* The bridge writes this generation before it enters the asynchronous
     * backend, so a fast completion cannot race the interface's filter state. */
    _activeScanGeneration = 0;
    setScanningState(request.identifier, true, nullptr, 0);

    IOReturn status = _bridge.startScan(selected ? channels : nullptr, selected,
                                        &_activeScanGeneration);
    if (status != kIOReturnSuccess) {
        _scanActive = false;
        _activeScanGeneration = 0;
        setScanningState(request.identifier, false, nullptr, 0);
    }
    return status;
}

IOReturn RTW88WCLInfraInterface::setWCL_ASSOCIATE(
    apple80211AssocCandidates *data)
{
    if (!data) return kIOReturnBadArgument;
    if (!_bridge.isAttached() || _stopping) return kIOReturnNotReady;

    const uint8_t *raw = reinterpret_cast<const uint8_t *>(data);
    const uint32_t candidates = RTW88WCLRequest::read32(raw + 0x214);
    if (candidates == 0 || candidates > 10) return kIOReturnBadArgument;
    const size_t requestLength = 0x218 + static_cast<size_t>(candidates) * 18;
    RTW88WCLRequest::Join request = {};
    if (!RTW88WCLRequest::decodeJoin(data, requestLength, request))
        return kIOReturnUnsupported;

    const uint8_t *pmk = request.keyLength == 32 ? request.key : nullptr;
    const uint8_t *rsn = request.rsnLength ? request.rsn : nullptr;
    _assocNotified = false;
    _connectNotified = false;
    memcpy(_targetBSSID, request.bssid, sizeof(_targetBSSID));
    /* The bridge writes this generation before cmdConnect() can schedule
     * authentication or produce an event. Duplicate pending joins reuse the
     * existing generation. */
    _activeJoinGeneration = 0;
    IOReturn status = _bridge.associate(request.ssid,
                                        static_cast<uint8_t>(request.ssidLength),
                                        request.bssid, pmk, rsn,
                                        request.rsnLength, false /* internal RTW88 supplicant */,
                                        &_activeJoinGeneration);
    bzero(request.key, sizeof(request.key));
    if (status != kIOReturnSuccess) {
        _activeJoinGeneration = 0;
        bzero(_targetBSSID, sizeof(_targetBSSID));
        return status;
    }
    return kIOReturnSuccess;
}

IOReturn RTW88WCLInfraInterface::setWCL_LINK_STATE_UPDATE(
    apple80211_wcl_update_link_state *data)
{
    if (!data) return kIOReturnBadArgument;

    /* Sequoia 24H32 WCL wire layout: byte 6 indicates an up-state record,
     * byte 8 says the BSSID field is valid, byte 7 indicates link-up, and
     * bytes 12..15 carry the down-state debounce interval. These offsets
     * match the reviewed 24H32 WCL protocol layout; don't cast the opaque
     * private structure to an older apple80211 definition. */
    const uint8_t *bytes = reinterpret_cast<const uint8_t *>(data);
    if (bytes[6]) {
        if (bytes[8])
            setCurrentApAddress(reinterpret_cast<ether_addr *>(data));
        if (bytes[7])
            setLinkState(kIO80211NetworkLinkUp, 0, false, 0);
        return kIOReturnSuccess;
    }

    const uint32_t debounce = RTW88WCLRequest::read32(bytes + 12);
    setCurrentApAddress(nullptr);
    setLinkState(kIO80211NetworkLinkDown, 0, debounce != 0, debounce);
    return kIOReturnSuccess;
}

bool RTW88WCLInfraInterface::scanResultMatches(const RTW88BSS &bss) const
{
    if (bss.channel == 0 || bss.channel > 196 || bss.ssid_len > 32 ||
        bss.ies_len > sizeof(bss.ies) ||
        !validInformationElements(bss.ies, bss.ies_len)) return false;
    if (_scanFilter.filterBSSID &&
        memcmp(bss.bssid, _scanFilter.bssid, sizeof(_scanFilter.bssid)) != 0)
        return false;
    if (_scanFilter.ssidLength &&
        (bss.ssid_len != _scanFilter.ssidLength ||
         memcmp(bss.ssid, _scanFilter.ssid, _scanFilter.ssidLength) != 0))
        return false;
    if (_scanFilter.filterChannels &&
        !(_scanFilter.channelBitmap[bss.channel / 8] &
          (1U << (bss.channel % 8))))
        return false;
    return true;
}

void RTW88WCLInfraInterface::publishScanResults(IOReturn status)
{
    if (!_controller) return;
    if (status == kIOReturnSuccess) {
        for (uint32_t index = 0; index < 128; ++index) {
            RTW88BSS bss = {};
            IOReturn copied = _bridge.copyScanBSS(index, &bss);
            if (copied == kIOReturnNotFound) break;
            if (copied != kIOReturnSuccess) break;
            if (!scanResultMatches(bss)) continue;

            uint8_t message[sizeof(RTW88WCLBeaconMetadata) + sizeof(bss.ies)] = {};
            RTW88WCLBeaconMetadata metadata = {};
            metadata.ieLength = bss.ies_len;
            metadata.channelSpec = static_cast<uint16_t>(bss.channel |
                (bss.channel <= 14 ? 0x1000 : 0xd000));
            memcpy(metadata.ssid, bss.ssid, bss.ssid_len);
            metadata.ssidLength = bss.ssid_len;
            metadata.channel = bss.channel;
            memcpy(metadata.bssid, bss.bssid, sizeof(metadata.bssid));
            metadata.rssi = bss.rssi;
            metadata.interval = bss.beacon_interval;
            metadata.capability = static_cast<uint16_t>(bss.capabilities);
            metadata.flags = kWCLBeaconRSSIValid;
            if (bss.ssid_len) metadata.flags |= kWCLBeaconSSIDPresent;
            memcpy(message, &metadata, sizeof(metadata));
            if (bss.ies_len)
                memcpy(message + sizeof(metadata), bss.ies, bss.ies_len);
            postWCLMessage(kWCLMessageScanResult, message,
                           sizeof(metadata) + bss.ies_len);
        }
    }

    _scanActive = false;
    _activeScanGeneration = 0;
    setScanningState(_scanFilter.identifier, false, nullptr, 0);
    postWCLMessage(kWCLMessageScanDone, &status, sizeof(status));
}

void RTW88WCLInfraInterface::postConnectionComplete(uint16_t status,
                                                     uint16_t reason,
                                                     IOReturn driverStatus)
{
    RTW88WCLConnectionEvent event = {};
    event.status = status;
    event.reason = reason;
    memcpy(event.peers[0].bssid, _targetBSSID, sizeof(_targetBSSID));
    event.peers[0].driverStatus = driverStatus;
    postWCLMessage(kWCLMessageConnectComplete, &event, sizeof(event));
}

void RTW88WCLInfraInterface::maybePublishConnected()
{
    /* setWCL_LINK_UP_DONE may be delivered after a leave/abort request. Never
     * publish an old backend CONNECTED state without a live join transaction. */
    if (!_activeJoinGeneration || _connectNotified || !_bridge.isAttached() ||
        !_bridge.connectionReady())
        return;

    _connectNotified = true;
    _assocNotified = true;
    postConnectionComplete(0, 0, kIOReturnSuccess);
    setLinkState(kIO80211NetworkLinkUp, 0, false, 0);

    RTW88WCLLinkStatusEvent link = {};
    memcpy(link.bssid, _targetBSSID, sizeof(link.bssid));
    link.up = 1;
    postWCLMessage(kWCLMessageLinkStatus, &link, sizeof(link));
}

void RTW88WCLInfraInterface::handleRTW88WCLRadioEvent(
    const RTW88WCLRadioEvent &event)
{
    if (_stopping) return;
    switch (event.event) {
        case kRTW88EventScanDone: {
            if (!_scanActive || event.scanGeneration != _activeScanGeneration)
                return; // stale completion from an aborted/superseded scan
            const IOReturn status = event.value ? kIOReturnAborted
                                                 : kIOReturnSuccess;
            publishScanResults(status);
            break;
        }
        case kRTW88EventConnected: {
            if (!_activeJoinGeneration ||
                event.joinGeneration != _activeJoinGeneration)
                return;
            if (!_assocNotified) {
                RTW88WCLAssociationEvent assoc = {};
                memcpy(assoc.bssid, _targetBSSID, sizeof(assoc.bssid));
                postWCLMessage(kWCLMessageAssocComplete, &assoc, sizeof(assoc));
                _assocNotified = true;
            }
            maybePublishConnected();
            break;
        }
        case kRTW88EventAssocDone: {
            if (!_activeJoinGeneration ||
                event.joinGeneration != _activeJoinGeneration)
                return; // stale association callback from an earlier join
            if (!_assocNotified) {
                RTW88WCLAssociationEvent assoc = {};
                memcpy(assoc.bssid, _targetBSSID, sizeof(assoc.bssid));
                postWCLMessage(kWCLMessageAssocComplete, &assoc, sizeof(assoc));
                _assocNotified = true;
            }
            maybePublishConnected();
            break;
        }
        case kRTW88EventDeauth: {
            if (!_activeJoinGeneration ||
                event.joinGeneration != _activeJoinGeneration)
                return;
            uint32_t reason = event.value;
            postWCLMessage(kWCLMessageDeauth, &reason, sizeof(reason));
            postConnectionComplete(1, static_cast<uint16_t>(reason),
                                   kIOReturnNotAttached);
            _connectNotified = false;
            _assocNotified = false;
            _activeJoinGeneration = 0;
            setCurrentApAddress(nullptr);
            setLinkState(kIO80211NetworkLinkDown, reason, false, 0);
            break;
        }
        case kRTW88EventDisconnected: {
            if (!_activeJoinGeneration ||
                event.joinGeneration != _activeJoinGeneration)
                return;
            if (_connectNotified || _assocNotified)
                postConnectionComplete(1, 0, kIOReturnNotAttached);
            _connectNotified = false;
            _assocNotified = false;
            _activeJoinGeneration = 0;
            setCurrentApAddress(nullptr);
            setLinkState(kIO80211NetworkLinkDown, 0, false, 0);
            RTW88WCLLinkStatusEvent link = {};
            memcpy(link.bssid, _targetBSSID, sizeof(link.bssid));
            link.up = 0;
            postWCLMessage(kWCLMessageLinkStatus, &link, sizeof(link));
            break;
        }
        case kRTW88EventRSSIChanged: {
            if (!_activeJoinGeneration ||
                event.joinGeneration != _activeJoinGeneration)
                return;
            int rssi = 0;
            if (_bridge.getRSSI(&rssi) == kIOReturnSuccess) {
                RTW88WCLLqmUpdate update = {};
                update.rssiValid = 1;
                update.rssi = rssi;
                postWCLMessage(kWCLMessageLqmUpdate, &update, sizeof(update));
            }
            break;
        }
    }
}

/* SPDX-License-Identifier: GPL-2.0-only
 * Fixed-capacity lock-free MPMC ring for owned pointers (mbufs or Skywalk packets).
 *
 * The ring does not allocate after reset(). It is safe for concurrent producers
 * and one or more consumers. reset() is lifecycle-only and must run after every
 * producer and consumer has been quiesced. A successful push transfers pointer
 * ownership to the ring; a failed push leaves ownership with the caller.
 */
#pragma once

#include <stdint.h>
#include <stddef.h>

// The bounded sequence algorithm uses a power-of-two capacity for masking.
template <typename T, uint32_t Capacity>
class RTW88WCLMbufRing final {
    static_assert(Capacity >= 2, "RX ring capacity must be at least two");
    static_assert((Capacity & (Capacity - 1)) == 0,
                  "RX ring capacity must be a power of two");

    struct Cell {
        volatile size_t sequence;
        T value;
    };

public:
    void reset()
    {
        for (size_t i = 0; i < Capacity; ++i) {
            _cells[i].value = T{};
            __atomic_store_n(&_cells[i].sequence, i, __ATOMIC_RELAXED);
        }
        __atomic_store_n(&_enqueuePosition, 0, __ATOMIC_RELAXED);
        __atomic_store_n(&_dequeuePosition, 0, __ATOMIC_RELAXED);
    }

    bool push(const T &value)
    {
        if (!value)
            return false;

        size_t position = __atomic_load_n(&_enqueuePosition, __ATOMIC_RELAXED);
        Cell *cell = nullptr;
        for (;;) {
            cell = &_cells[position & (Capacity - 1)];
            const size_t sequence = __atomic_load_n(&cell->sequence,
                                                    __ATOMIC_ACQUIRE);
            const intptr_t difference = static_cast<intptr_t>(sequence) -
                                       static_cast<intptr_t>(position);
            if (difference == 0) {
                size_t expected = position;
                if (__atomic_compare_exchange_n(&_enqueuePosition, &expected,
                                                position + 1, false,
                                                __ATOMIC_RELAXED,
                                                __ATOMIC_RELAXED))
                    break;
                position = expected;
            } else if (difference < 0) {
                return false; // bounded queue full; caller keeps ownership
            } else {
                position = __atomic_load_n(&_enqueuePosition,
                                           __ATOMIC_RELAXED);
            }
        }

        cell->value = value;
        __atomic_store_n(&cell->sequence, position + 1, __ATOMIC_RELEASE);
        return true;
    }

    bool pop(T *out)
    {
        if (!out)
            return false;

        size_t position = __atomic_load_n(&_dequeuePosition, __ATOMIC_RELAXED);
        Cell *cell = nullptr;
        for (;;) {
            cell = &_cells[position & (Capacity - 1)];
            const size_t sequence = __atomic_load_n(&cell->sequence,
                                                    __ATOMIC_ACQUIRE);
            const intptr_t difference = static_cast<intptr_t>(sequence) -
                                       static_cast<intptr_t>(position + 1);
            if (difference == 0) {
                size_t expected = position;
                if (__atomic_compare_exchange_n(&_dequeuePosition, &expected,
                                                position + 1, false,
                                                __ATOMIC_RELAXED,
                                                __ATOMIC_RELAXED))
                    break;
                position = expected;
            } else if (difference < 0) {
                return false; // empty or a producer has not published yet
            } else {
                position = __atomic_load_n(&_dequeuePosition,
                                           __ATOMIC_RELAXED);
            }
        }

        *out = cell->value;
        cell->value = T{};
        __atomic_store_n(&cell->sequence, position + Capacity,
                         __ATOMIC_RELEASE);
        return true;
    }

    bool empty() const
    {
        const size_t position = __atomic_load_n(&_dequeuePosition,
                                                __ATOMIC_RELAXED);
        const Cell *cell = &_cells[position & (Capacity - 1)];
        const size_t sequence = __atomic_load_n(&cell->sequence,
                                                __ATOMIC_ACQUIRE);
        const intptr_t difference = static_cast<intptr_t>(sequence) -
                                   static_cast<intptr_t>(position + 1);
        return difference < 0;
    }

    uint32_t approximateSize() const
    {
        const size_t enqueued = __atomic_load_n(&_enqueuePosition,
                                               __ATOMIC_ACQUIRE);
        const size_t dequeued = __atomic_load_n(&_dequeuePosition,
                                                __ATOMIC_ACQUIRE);
        const size_t count = enqueued >= dequeued ? enqueued - dequeued : 0;
        return count > Capacity ? Capacity : static_cast<uint32_t>(count);
    }

    static constexpr uint32_t capacity() { return Capacity; }

private:
    alignas(64) volatile size_t _enqueuePosition = 0;
    alignas(64) volatile size_t _dequeuePosition = 0;
    alignas(64) Cell _cells[Capacity] = {};
};

IOReturn RTW88WCLPCIBackend::open(IOPCIDevice *provider,
                                  IOWorkLoop *workLoop,
                                  RTW88WCLPCIBackendClient *client)
{
    if (!provider || !workLoop)
        return kIOReturnBadArgument;
    if (_opened || _pciDev || _compatLeaseHeld)
        return kIOReturnBusy;

    _pciDev = provider;
    _pciDev->retain();
    _workLoop = workLoop;
    _workLoop->retain();
    _client = client;

    _dmaLock = IOSimpleLockAlloc();
    _pendingFreeLock = IOSimpleLockAlloc();
    if (!_dmaLock || !_pendingFreeLock) {
        closeResources();
        return kIOReturnNoMemory;
    }

    _pciDev->setMemoryEnable(true);
    _pciDev->setBusMasterEnable(true);
    const UInt16 enabledCommand = _pciDev->configRead16(kIOPCIConfigCommand);
    if ((enabledCommand & 0x0006U) != 0x0006U) {
        IOLog("rtl88wifi-wcl: PCI memory/bus-master enable failed (command=0x%04x)\n",
              enabledCommand);
        closeResources();
        return kIOReturnNotReady;
    }

    /* rtw88's PCI backend maps BAR2 (the Linux driver uses bar_id=2). */
    _mmioMap = _pciDev->mapDeviceMemoryWithRegister(kIOPCIConfigBaseAddress2);
    if (!_mmioMap || !_mmioMap->getVirtualAddress() || !_mmioMap->getLength()) {
        IOLog("rtl88wifi-wcl: unable to map a usable BAR2\n");
        closeResources();
        return kIOReturnNoResources;
    }
    _mmioBase = reinterpret_cast<volatile void *>(_mmioMap->getVirtualAddress());

    _compatPciDev = static_cast<struct pci_dev *>(IOMallocZero(sizeof(*_compatPciDev)));
    if (!_compatPciDev) {
        closeResources();
        return kIOReturnNoMemory;
    }
    _compatPciDev->vendor = _pciDev->configRead16(0x00);
    _compatPciDev->device = _pciDev->configRead16(0x02);
    _compatPciDev->kext_dev = this;
    _compatPciDev->resource[2] = reinterpret_cast<resource_size_t>(_mmioBase);
    _compatPciDev->resource_len[2] = static_cast<resource_size_t>(_mmioMap->getLength());

    /* The compat layer has process-global callbacks, so acquire this lease
     * before publishing the function tables or starting its workqueues. */
    RTW88HwOps *expectedOwner = nullptr;
    if (!__atomic_compare_exchange_n(&g_pci_dev_instance, &expectedOwner,
                                     static_cast<RTW88HwOps *>(this), false,
                                     __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE)) {
        IOLog("rtl88wifi-wcl: RTW88 compat hardware lease is already owned\n");
        closeResources();
        return kIOReturnBusy;
    }
    _compatLeaseHeld = true;
    rtw88_pci_io_ops = &_pci_io_ops;
    rtw88_dma_ops = &_dma_ops;

    const int compatStatus = rtw88_compat_init();
    if (compatStatus != 0) {
        IOLog("rtl88wifi-wcl: compat runtime initialization failed (%d)\n",
              compatStatus);
        closeResources();
        return kIOReturnNoResources;
    }
    _compatInitialized = true;

    _interruptSource = IOInterruptEventSource::interruptEventSource(
        this,
        OSMemberFunctionCast(IOInterruptEventSource::Action, this,
                             &RTW88WCLPCIBackend::handleInterrupt),
        _pciDev, 0);
    if (!_interruptSource ||
        _workLoop->addEventSource(_interruptSource) != kIOReturnSuccess) {
        IOLog("rtl88wifi-wcl: unable to register PCI interrupt event source\n");
        closeResources();
        return kIOReturnNoResources;
    }
    _interruptSource->disable();

    _opened = true;
    IOLog("rtl88wifi-wcl: PCI backend acquired %04x:%04x, BAR2=%p length=0x%llx\n",
          _compatPciDev->vendor, _compatPciDev->device, (void *)_mmioBase,
          static_cast<unsigned long long>(_mmioMap->getLength()));
    return kIOReturnSuccess;
}

IOReturn RTW88WCLPCIBackend::setRadioRunning(bool running)
{
    if (!_opened || !_pciDev || !_interruptSource)
        return kIOReturnNotReady;
    if (_radioRunning == running)
        return kIOReturnSuccess;

    if (running) {
        _pciDev->setMemoryEnable(true);
        _pciDev->setBusMasterEnable(true);
        const UInt16 command = _pciDev->configRead16(kIOPCIConfigCommand);
        if ((command & 0x0006U) != 0x0006U) {
            _interruptSource->disable();
            _pciDev->setBusMasterEnable(false);
            const UInt16 rollbackCommand = _pciDev->configRead16(kIOPCIConfigCommand);
            if (rollbackCommand & 0x0004U) {
                /* Bus mastering may still be active. Keep the backend marked
                 * running so close/free cannot reclaim DMA mappings. */
                _radioRunning = true;
                IOLog("rtl88wifi-wcl: partial PCI enable rollback failed (command=0x%04x); DMA resources quarantined\n",
                      rollbackCommand);
            } else {
                _radioRunning = false;
                _pciDev->setMemoryEnable(false);
            }
            return kIOReturnNotReady;
        }
        _radioRunning = true;
        _interruptSource->enable();
        return kIOReturnSuccess;
    }

    /* Caller contract: RTW88IEEE80211::stop() must already have synchronized
     * its interrupt/NAPI workers and stopped device DMA before this call. */
    _interruptSource->disable();
    _pciDev->setBusMasterEnable(false);
    const UInt16 command = _pciDev->configRead16(kIOPCIConfigCommand);
    if (command & 0x0004U) {
        IOLog("rtl88wifi-wcl: PCI bus-master fence did not stick (command=0x%04x)\n",
              command);
        return kIOReturnNotReady;
    }
    _radioRunning = false;
    if (preemption_enabled())
        drainPendingFree();
    return kIOReturnSuccess;
}

IOReturn RTW88WCLPCIBackend::close()
{
    if (_radioRunning)
        return kIOReturnBusy;
    closeResources();
    return kIOReturnSuccess;
}

void RTW88WCLPCIBackend::closeResources()
{
    if (_interruptSource) {
        _interruptSource->disable();
        if (_workLoop)
            _workLoop->removeEventSource(_interruptSource);
        _interruptSource->release();
        _interruptSource = nullptr;
    }

    /* Stop all compat work before reclaiming DMA buffers and releasing the
     * callbacks they may invoke. The RTW88 core must already be stopped. */
    if (_compatInitialized) {
        rtw88_compat_exit();
        _compatInitialized = false;
    }

    if (preemption_enabled())
        drainPendingFree();
    releaseDMAEntries();

    clearCompatLease();

    if (_pciDev) {
        _pciDev->setBusMasterEnable(false);
        _pciDev->setMemoryEnable(false);
    }
    if (_compatPciDev) {
        IOFree(_compatPciDev, sizeof(*_compatPciDev));
        _compatPciDev = nullptr;
    }
    if (_mmioMap) {
        _mmioMap->release();
        _mmioMap = nullptr;
    }
    _mmioBase = nullptr;

    if (_workLoop) {
        _workLoop->release();
        _workLoop = nullptr;
    }
    if (_pciDev) {
        _pciDev->release();
        _pciDev = nullptr;
    }
    if (_pendingFreeLock) {
        IOSimpleLockFree(_pendingFreeLock);
        _pendingFreeLock = nullptr;
    }
    if (_dmaLock) {
        IOSimpleLockFree(_dmaLock);
        _dmaLock = nullptr;
    }

    _client = nullptr;
    _radioRunning = false;
    _opened = false;
}

void RTW88WCLPCIBackend::clearCompatLease()
{
    if (!_compatLeaseHeld)
        return;
    RTW88HwOps *expectedOwner = static_cast<RTW88HwOps *>(this);
    if (__atomic_compare_exchange_n(&g_pci_dev_instance, &expectedOwner,
                                    static_cast<RTW88HwOps *>(nullptr), false,
                                    __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE)) {
        rtw88_pci_io_ops = nullptr;
        rtw88_dma_ops = nullptr;
    }
    _compatLeaseHeld = false;
}

void RTW88WCLPCIBackend::free()
{
    if (_pciDev) {
        if (_interruptSource)
            _interruptSource->disable();
        _pciDev->setBusMasterEnable(false);
        const UInt16 command = _pciDev->configRead16(kIOPCIConfigCommand);
        if (command & 0x0004U) {
            IOLog("rtl88wifi-wcl: PCI bus-master bit remains set (command=0x%04x); refusing to free DMA resources; backend quarantined\n",
                  command);
            /* Keep this object, DMA descriptors, MMIO map, locks, and PCI
             * reference alive until hardware bus mastering is demonstrably off. */
            return;
        }
    }
    _radioRunning = false;
    closeResources();
    super::free();
}

void *RTW88WCLPCIBackend::allocCoherent(size_t size,
                                         IOPhysicalAddress *physical)
{
    if (physical) *physical = 0;
    if (!size || !_compatLeaseHeld || !_dmaLock || !_pendingFreeLock)
        return nullptr;
    if (preemption_enabled())
        drainPendingFree();

    IOBufferMemoryDescriptor *descriptor =
        IOBufferMemoryDescriptor::inTaskWithPhysicalMask(
            kernel_task,
            kIOMemoryPhysicallyContiguous | kIODirectionInOut |
                kIOMemoryKernelUserShared,
            size, 0x00000000FFFFFFF0ULL);
    if (!descriptor)
        return nullptr;
    if (descriptor->prepare() != kIOReturnSuccess) {
        descriptor->release();
        return nullptr;
    }

    const IOPhysicalAddress address = descriptor->getPhysicalAddress();
    void *virtualAddress = descriptor->getBytesNoCopy();
    if (!virtualAddress || !address) {
        descriptor->complete();
        descriptor->release();
        return nullptr;
    }
    bzero(virtualAddress, size);

    RTW88DMAEntry *entry = static_cast<RTW88DMAEntry *>(IOMallocZero(sizeof(*entry)));
    if (!entry) {
        descriptor->complete();
        descriptor->release();
        return nullptr;
    }
    entry->desc = descriptor;
    entry->virt = virtualAddress;
    entry->phys = address;
    entry->size = size;

    IOSimpleLockLock(_dmaLock);
    entry->next = _dmaList;
    _dmaList = entry;
    IOSimpleLockUnlock(_dmaLock);
    if (physical) *physical = address;
    return virtualAddress;
}

void RTW88WCLPCIBackend::freeCoherent(size_t size, void *virtualAddress,
                                      IOPhysicalAddress physical)
{
    (void)size;
    (void)physical;
    if (!_dmaLock || !_pendingFreeLock || !virtualAddress)
        return;

    IOSimpleLockLock(_dmaLock);
    RTW88DMAEntry **previous = &_dmaList;
    RTW88DMAEntry *entry = _dmaList;
    while (entry && entry->virt != virtualAddress) {
        previous = &entry->next;
        entry = entry->next;
    }
    if (entry)
        *previous = entry->next;
    IOSimpleLockUnlock(_dmaLock);

    if (!entry) {
        IOLog("rtl88wifi-wcl: freeCoherent: unknown virtual address %p\n",
              virtualAddress);
        return;
    }
    if (preemption_enabled()) {
        freeDMAList(entry);
    } else {
        IOSimpleLockLock(_pendingFreeLock);
        entry->next = _dmaPendingFree;
        _dmaPendingFree = entry;
        IOSimpleLockUnlock(_pendingFreeLock);
    }
}

void RTW88WCLPCIBackend::freeCoherentByPhys(IOPhysicalAddress physical)
{
    if (!_dmaLock || !_pendingFreeLock || !physical)
        return;

    IOSimpleLockLock(_dmaLock);
    RTW88DMAEntry **previous = &_dmaList;
    RTW88DMAEntry *entry = _dmaList;
    while (entry && entry->phys != physical) {
        previous = &entry->next;
        entry = entry->next;
    }
    if (entry)
        *previous = entry->next;
    IOSimpleLockUnlock(_dmaLock);

    if (!entry)
        return;
    if (preemption_enabled()) {
        freeDMAList(entry);
    } else {
        IOSimpleLockLock(_pendingFreeLock);
        entry->next = _dmaPendingFree;
        _dmaPendingFree = entry;
        IOSimpleLockUnlock(_pendingFreeLock);
    }
}

void RTW88WCLPCIBackend::setBounceOrigVA(IOPhysicalAddress physical,
                                         void *originalAddress)
{
    if (!_dmaLock || !physical)
        return;
    IOSimpleLockLock(_dmaLock);
    for (RTW88DMAEntry *entry = _dmaList; entry; entry = entry->next) {
        if (entry->phys == physical) {
            entry->orig_va = originalAddress;
            break;
        }
    }
    IOSimpleLockUnlock(_dmaLock);
}

void RTW88WCLPCIBackend::syncBounceForCpu(IOPhysicalAddress dmaAddress,
                                          size_t size)
{
    if (!_dmaLock || !dmaAddress || !size)
        return;
    /* Keep the entry pinned by the list lock for the bounded copy; this avoids
     * racing a concurrent unmap/free against the original buffer pointer. */
    IOSimpleLockLock(_dmaLock);
    for (RTW88DMAEntry *entry = _dmaList; entry; entry = entry->next) {
        if (entry->phys == dmaAddress && entry->orig_va && entry->virt) {
            const size_t copyLength = size < entry->size ? size : entry->size;
            memcpy(entry->orig_va, entry->virt, copyLength);
            break;
        }
    }
    IOSimpleLockUnlock(_dmaLock);
}

void RTW88WCLPCIBackend::resumeTxIfStalled()
{
    if (_client)
        _client->rtw88WCLResumeTransmit();
}

void RTW88WCLPCIBackend::handleInterrupt(IOInterruptEventSource *source,
                                          int count)
{
    (void)source;
    (void)count;
    if (!_radioRunning || !g_pci_dev_instance ||
        __atomic_load_n(&g_pci_dev_instance, __ATOMIC_ACQUIRE) !=
            static_cast<RTW88HwOps *>(this))
        return;
    if (preemption_enabled())
        drainPendingFree();
    rtw88_trigger_interrupt();
}

void RTW88WCLPCIBackend::drainPendingFree()
{
    if (!_pendingFreeLock)
        return;
    IOSimpleLockLock(_pendingFreeLock);
    RTW88DMAEntry *list = _dmaPendingFree;
    _dmaPendingFree = nullptr;
    IOSimpleLockUnlock(_pendingFreeLock);
    freeDMAList(list);
}

void RTW88WCLPCIBackend::releaseDMAEntries()
{
    if (_dmaLock) {
        IOSimpleLockLock(_dmaLock);
        RTW88DMAEntry *list = _dmaList;
        _dmaList = nullptr;
        IOSimpleLockUnlock(_dmaLock);
        freeDMAList(list);
    }
    if (_pendingFreeLock) {
        IOSimpleLockLock(_pendingFreeLock);
        RTW88DMAEntry *pending = _dmaPendingFree;
        _dmaPendingFree = nullptr;
        IOSimpleLockUnlock(_pendingFreeLock);
        freeDMAList(pending);
    }
}

void RTW88WCLPCIBackend::freeDMAList(RTW88DMAEntry *list)
{
    while (list) {
        RTW88DMAEntry *next = list->next;
        if (list->desc) {
            list->desc->complete();
            list->desc->release();
        }
        IOFree(list, sizeof(*list));
        list = next;
    }
}

static inline bool RTW88WCLPacketSpanIsValid(const RTW88WCLPacketSpan &span)

static inline bool RTW88WCLResolvePacketBufferAddress(

{
    if (!address || segmentBase == 0 ||
        !RTW88WCLPacketSpanIsValid(span))
        return false;

    if (span.bufferOffset > UINT64_MAX - span.dataOffset)
        return false;
    const uint64_t relative = span.bufferOffset + span.dataOffset;
    if (segmentBase > UINT64_MAX - relative)
        return false;

    const uint64_t candidate = segmentBase + relative;
    if (candidate > static_cast<uint64_t>(UINTPTR_MAX) ||
        span.dataLength > static_cast<uint64_t>(UINTPTR_MAX) - candidate)
        return false;

    *address = static_cast<uintptr_t>(candidate);
    return true;
}

     * hand us a chained mbuf; mbuf_copydata() below safely spans the chain. */

bool RTW88WCLSkywalkRxAdapter::copyMbufToSkywalkPacket(
    IOSkywalkPacketBufferPool *pool, mbuf_t source,
    IOSkywalkPacket **outPacket)
{
    if (outPacket)
        *outPacket = nullptr;
    if (!pool || !source || !outPacket)
        return false;

    const size_t packetLength = mbuf_pkthdr_len(source);
    if (packetLength < kMinimumFrameLength ||
        packetLength > kMaximumFrameLength)
        return false;

    IOSkywalkPacket *packet = nullptr;
    /* Option bit 0 requests non-blocking allocation in a packet callback. */
    if (pool->allocatePacket(1, &packet, 1) != kIOReturnSuccess || !packet)
        return false;

    bool valid = false;
    do {
        if (packet->getPacketType() != kIOSkywalkPacketTypeNetwork ||
            packet->getPacketBufferPool() != pool ||
            packet->getPacketBufferCount() != 1)
            break;

        auto *networkPacket = OSDynamicCast(IOSkywalkNetworkPacket, packet);
        if (!networkPacket)
            break;

        IOSkywalkPacketBuffer *buffers[1] = { nullptr };
        if (packet->getPacketBuffers(buffers, 1) != 1 || !buffers[0])
            break;
        IOSkywalkPacketBuffer *buffer = buffers[0];
        if (buffer->getPacketBufferPool() != pool)
            break;

        IOSkywalkMemorySegment *segment = buffer->getMemorySegment();
        if (!segment)
            break;
        IOMemoryDescriptor *memory = segment->getMemoryDescriptor();
        if (!memory)
            break;

        const uint64_t segmentBase = segment->getVirtualAddress();
        const RTW88WCLPacketSpan span = {
            memory->getLength(),
            buffer->getMemorySegmentOffset(),
            buffer->getBufferSize(),
            0,
            static_cast<uint32_t>(packetLength)
        };
        uintptr_t destinationAddress = 0;
        if (!RTW88WCLResolvePacketBufferAddress(
                segmentBase, span, &destinationAddress))
            break;

        if (packet->setDataOffsetAndLength(
                0, static_cast<UInt32>(packetLength)) != kIOReturnSuccess)
            break;
        if (packet->getDataOffset() != 0 ||
            packet->getDataLength() != packetLength ||
            buffer->getDataOffset() != 0 ||
            buffer->getDataLength() != packetLength)
            break;

        auto *destination = reinterpret_cast<void *>(destinationAddress);
        if (mbuf_copydata(source, 0, packetLength, destination) != 0)
            break;

        if (networkPacket->setCompletionStatus(0) != kIOReturnSuccess)
            break;

        *outPacket = packet;
        valid = true;
    } while (false);

    if (!valid)
        (void)pool->deallocatePacket(packet);
    return valid;
}

IOReturn RTW88WCLSkywalkRxAdapter::enqueueFrame(mbuf_t packet)
{
    if (!packet)
        return kIOReturnBadArgument;

    if (__atomic_load_n(&_stopping, __ATOMIC_ACQUIRE) ||
        !isValidMbuf(packet)) {
        __atomic_add_fetch(&_dropped, 1U, __ATOMIC_RELAXED);
        mbuf_freem(packet);
        return kIOReturnNotReady;
    }

    if (!_frames.push(packet)) {
        __atomic_add_fetch(&_dropped, 1U, __ATOMIC_RELAXED);
        mbuf_freem(packet);
        return kIOReturnNoResources;
    }

    uint32_t expected = 0;
    if (__atomic_compare_exchange_n(&_pendingSignal, &expected, 1U, false,
                                    __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE))
        signalWorkAvailable();
    return kIOReturnSuccess;
}

bool RTW88WCLSkywalkRxAdapter::checkForWork()
{
    if (__atomic_load_n(&_stopping, __ATOMIC_ACQUIRE))
        return false;

    uint32_t drained = 0;
    mbuf_t source = nullptr;
    while (drained < kDrainBudget && _frames.pop(&source)) {
        if (source) {
            IOSkywalkPacket *packet = nullptr;
            if (copyMbufToSkywalkPacket(_rxPool, source, &packet)) {
                __atomic_add_fetch(&_prepared, 1U, __ATOMIC_RELAXED);
                /* A true return means the sink has accepted ownership. The
                 * current controller sink returns false because the RX
                 * completion queue is still unregistered. */
                const bool accepted = _sink && _sink(packet, _sinkContext);
                if (!accepted) {
                    (void)_rxPool->deallocatePacket(packet);
                    __atomic_add_fetch(&_dropped, 1U, __ATOMIC_RELAXED);
                }
            } else {
                __atomic_add_fetch(&_dropped, 1U, __ATOMIC_RELAXED);
            }
            mbuf_freem(source);
            source = nullptr;
        }
        ++drained;
    }

    if (drained == kDrainBudget)
        return true; // bounded workloop occupancy; schedule another pass

    __atomic_store_n(&_pendingSignal, 0U, __ATOMIC_RELEASE);
    if (!_frames.empty()) {
        uint32_t expected = 0;
        if (__atomic_compare_exchange_n(&_pendingSignal, &expected, 1U,
                                        false, __ATOMIC_ACQ_REL,
                                        __ATOMIC_ACQUIRE))
            return true;
    }
    return false;
}

void RTW88WCLSkywalkRxAdapter::stop()
{
    __atomic_store_n(&_stopping, 1U, __ATOMIC_RELEASE);
    if (_registered && _workLoop) {
        /* Synchronous removal serializes against checkForWork(). The owner
         * must stop/quiesce RTW88 RX producers before invoking stop(). */
        _workLoop->removeEventSource(this);
        _registered = false;
    }

    mbuf_t packet = nullptr;
    while (_frames.pop(&packet)) {
        if (packet)
            mbuf_freem(packet);
        packet = nullptr;
    }
    _sink = nullptr;
    _sinkContext = nullptr;
    _workLoop = nullptr;
    if (_rxPool) {
        _rxPool->release();
        _rxPool = nullptr;
    }
}

void RTW88WCLSkywalkRxAdapter::free()
{
    stop();
    super::free();
}

{
    if (_tickets.hasOutstanding())
        return kIOReturnBusy;

    RTW88IEEE80211 *radio = _radio;
    IOSkywalkPacketBufferPool *pool = _txPool;
    _radio = nullptr;
    _txPool = nullptr;
    if (pool)
        pool->release();
    if (radio)
        radio->release();
    return kIOReturnSuccess;
}

mbuf_t RTW88WCLSkywalkTxAdapter::copyPacketToMbuf(
    const uint8_t *bytes, uint32_t length)
{
    if (!bytes || length < 14 || length > 2048)
        return nullptr;

    mbuf_t mbuf = nullptr;
    if (mbuf_allocpacket(MBUF_DONTWAIT, length, nullptr, &mbuf) != 0 || !mbuf)
        return nullptr;

    /* allocpacket may return a chain; copyback handles every segment. Keep
     * the copy non-blocking because this runs in a Skywalk queue callback. */
    if (mbuf_copyback(mbuf, 0, length, bytes, MBUF_DONTWAIT) != 0) {
        mbuf_freem(mbuf);
        return nullptr;
    }
    mbuf_pkthdr_setlen(mbuf, length);
    return mbuf;
}

IOReturn RTW88WCLSkywalkTxAdapter::submitPacket(IOSkywalkPacket *packet)
{
    if (!packet)
        return kIOReturnBadArgument;
    if (!_radio || !_txPool)
        return kIOReturnNotReady;

    RTW88WCLPacketView view = {};
    if (!RTW88WCLResolveEthernetTxPacket(packet, _txPool, &view))
        return kIOReturnBadArgument;

    mbuf_t mbuf = copyPacketToMbuf(view.bytes, view.length);
    if (!mbuf)
        return kIOReturnNoMemory;

    packet->retain();
    TicketPool::Ticket *ticket = _tickets.reserve(packet);
    if (!ticket) {
        packet->release();
        mbuf_freem(mbuf);
        return kIOReturnBusy;
    }

    /* Keep the adapter alive until the workloop-side completion drain drops
     * this ticket. backendCompletion() intentionally does not release it. */
    retain();
    const UInt32 backendResult = _radio->outputPacketWithCompletion(
        mbuf, &RTW88WCLSkywalkTxAdapter::backendCompletion, this, ticket);

    /* outputPacketWithCompletion consumes the mbuf and reports exactly one
     * terminal callback, including synchronous drops. Once the ticket has
     * been reserved, adapter acceptance is successful; the actual transmitted
     * vs dropped result is delivered later through the completion sink. */
    (void)backendResult;
    return kIOReturnSuccess;
}

void RTW88WCLSkywalkTxAdapter::backendCompletion(
    void *context, void *cookie, bool transmitted)
{
    auto *self = static_cast<RTW88WCLSkywalkTxAdapter *>(context);
    auto *ticket = static_cast<TicketPool::Ticket *>(cookie);
    if (!self || !ticket)
        return;

    /* No Skywalk API, OSObject release, allocation, or blocking operation is
     * allowed here. A deferred consumer performs the completion handoff. */
    (void)TicketPool::complete(ticket, transmitted);
}

uint32_t RTW88WCLSkywalkTxAdapter::drainCompletions(
    CompletionSink sink, void *context)
{
    if (!sink)
        return 0;

    /* A sink may trigger owner teardown; keep this adapter alive until the
     * drain loop has stopped touching its ticket pool. */
    retain();
    uint32_t drained = 0;
    for (;;) {
        TicketPool::Ticket *ticket = _tickets.claimCompleted();
        if (!ticket)
            break;

        auto *packet = static_cast<IOSkywalkPacket *>(TicketPool::packet(ticket));
        const bool transmitted = TicketPool::transmitted(ticket);

        /* The sink runs in deferred context and borrows the retained packet.
         * It must finish any Skywalk queue handoff before returning. */
        if (packet)
            sink(packet, transmitted, context);

        TicketPool::finish(ticket);
        if (packet)
            packet->release();
        release();
        ++drained;
    }
    release();
    return drained;
}

bool RTW88WCLSkywalkTxAdapter::hasOutstandingCompletions() const
{
    return _tickets.hasOutstanding();
}

uint32_t RTW88WCLSkywalkTxAdapter::outstandingCompletionCount() const
{
    return _tickets.outstandingCount();
}

void RTW88WCLSkywalkTxAdapter::free()
{
    /* A per-ticket retain prevents free() while hardware callbacks are still
     * pending. The owning controller must drain every completion before its
     * final reference is released. */
    if (_tickets.hasOutstanding())
        IOLog("rtl88wifi-wcl: TX adapter free attempted with outstanding tickets\n");
    unbind();
    super::free();
}

/* SPDX-License-Identifier: GPL-2.0-only
 * Fixed-capacity completion tickets for the native WCL TX adapter.
 *
 * Multiple RTW88 completion contexts may publish concurrently. The Skywalk
 * packet operations are deliberately kept out of complete(); a workloop-side
 * consumer claims completed tickets and performs the deferred queue handoff.
 */
#pragma once

#include <stdint.h>

/* Freestanding so the state machine can be stress-tested outside the kernel. */
template <uint32_t Capacity>
class RTW88WCLTxCompletionSlots {
    static_assert(Capacity > 0, "completion slot capacity must be non-zero");

public:
    enum : uint32_t {
        kFree = 0,
        kReserved,
        kSubmitted,
        kCompleting,
        kCompleted,
        kDraining
    };

    struct Ticket {
        volatile uint32_t state;
        void *packet;
        uint32_t transmitted;
    };

    void reset()
    {
        for (uint32_t i = 0; i < Capacity; ++i) {
            _slots[i].packet = nullptr;
            _slots[i].transmitted = 0;
            __atomic_store_n(&_slots[i].state, kFree, __ATOMIC_RELAXED);
        }
    }

    /* Called before handing a packet to RTW88. The caller owns the packet
     * retain associated with the returned ticket. */
    Ticket *reserve(void *packet)
    {
        if (!packet)
            return nullptr;

        for (uint32_t i = 0; i < Capacity; ++i) {
            Ticket *ticket = &_slots[i];
            uint32_t expected = kFree;
            if (!__atomic_compare_exchange_n(&ticket->state, &expected,
                                             kReserved, false,
                                             __ATOMIC_ACQ_REL,
                                             __ATOMIC_ACQUIRE))
                continue;

            ticket->packet = packet;
            ticket->transmitted = 0;
            __atomic_store_n(&ticket->state, kSubmitted, __ATOMIC_RELEASE);
            return ticket;
        }
        return nullptr;
    }

    /* Safe for the RTW88 completion callback: bounded atomics only, no
     * allocation, locks, Skywalk calls, or packet release. */
    static bool complete(Ticket *ticket, bool transmitted)
    {
        if (!ticket)
            return false;
        uint32_t expected = kSubmitted;
        if (!__atomic_compare_exchange_n(&ticket->state, &expected,
                                         kCompleting, false,
                                         __ATOMIC_ACQ_REL,
                                         __ATOMIC_ACQUIRE))
            return false;
        ticket->transmitted = transmitted ? 1U : 0U;
        __atomic_store_n(&ticket->state, kCompleted, __ATOMIC_RELEASE);
        return true;
    }

    /* Single consumer, called from the adapter's deferred/workloop context. */
    Ticket *claimCompleted()
    {
        for (uint32_t i = 0; i < Capacity; ++i) {
            Ticket *ticket = &_slots[i];
            uint32_t expected = kCompleted;
            if (__atomic_compare_exchange_n(&ticket->state, &expected,
                                            kDraining, false,
                                            __ATOMIC_ACQ_REL,
                                            __ATOMIC_ACQUIRE))
                return ticket;
        }
        return nullptr;
    }

    static void finish(Ticket *ticket)
    {
        if (!ticket)
            return;
        ticket->packet = nullptr;
        ticket->transmitted = 0;
        __atomic_store_n(&ticket->state, kFree, __ATOMIC_RELEASE);
    }

    static void *packet(const Ticket *ticket)
    {
        return ticket ? ticket->packet : nullptr;
    }

    static bool transmitted(const Ticket *ticket)
    {
        return ticket && ticket->transmitted != 0;
    }

    bool hasOutstanding() const
    {
        for (uint32_t i = 0; i < Capacity; ++i) {
            if (__atomic_load_n(&_slots[i].state, __ATOMIC_ACQUIRE) != kFree)
                return true;
        }
        return false;
    }

    uint32_t outstandingCount() const
    {
        uint32_t count = 0;
        for (uint32_t i = 0; i < Capacity; ++i) {
            if (__atomic_load_n(&_slots[i].state, __ATOMIC_ACQUIRE) != kFree)
                ++count;
        }
        return count;
    }

    static constexpr uint32_t capacity() { return Capacity; }

private:
    Ticket _slots[Capacity] = {};
};
