00609:                                         request.bssid, pmk, rsn,

00628: IOReturn RTW88WCLInfraInterface::setWCL_LINK_STATE_UPDATE(
00629:     apple80211_wcl_update_link_state *data)
00630: {
00631:     if (!data) return kIOReturnBadArgument;
00632: 
00633:     /* Sequoia 24H32 WCL wire layout: byte 6 indicates an up-state record,
00634:      * byte 8 says the BSSID field is valid, byte 7 indicates link-up, and
00635:      * bytes 12..15 carry the down-state debounce interval. These offsets
00636:      * match the reviewed 24H32 WCL protocol layout; don't cast the opaque
00637:      * private structure to an older apple80211 definition. */

00639:     if (bytes[6]) {

00643:             setLinkState(kIO80211NetworkLinkUp, 0, false, 0);
00644:         return kIOReturnSuccess;
00645:     }

00647:     const uint32_t debounce = RTW88WCLRequest::read32(bytes + 12);

00653: bool RTW88WCLInfraInterface::scanResultMatches(const RTW88BSS &bss) const
00654: {
00655:     if (bss.channel == 0 || bss.channel > 196 || bss.ssid_len > 32 ||
00656:         bss.ies_len > sizeof(bss.ies) ||
00657:         !validInformationElements(bss.ies, bss.ies_len)) return false;
00658:     if (_scanFilter.filterBSSID &&
00659:         memcmp(bss.bssid, _scanFilter.bssid, sizeof(_scanFilter.bssid)) != 0)
00660:         return false;
00661:     if (_scanFilter.ssidLength &&
00662:         (bss.ssid_len != _scanFilter.ssidLength ||
00663:          memcmp(bss.ssid, _scanFilter.ssid, _scanFilter.ssidLength) != 0))
00664:         return false;
00665:     if (_scanFilter.filterChannels &&
00666:         !(_scanFilter.channelBitmap[bss.channel / 8] &
00667:           (1U << (bss.channel % 8))))
00668:         return false;
00669:     return true;
00670: }

00680: void RTW88WCLInfraInterface::publishScanResults(IOReturn status)
00681: {
00682:     if (!_controller) return;
00683:     if (status == kIOReturnSuccess) {
00684:         for (uint32_t index = 0; index < 128; ++index) {
00685:             RTW88BSS bss = {};
00686:             IOReturn copied = _bridge.copyScanBSS(index, &bss);
00687:             if (copied == kIOReturnNotFound) break;
00688:             if (copied != kIOReturnSuccess) break;
00689:             if (!scanResultMatches(bss)) continue;
00690: 
00691:             uint8_t message[sizeof(RTW88WCLBeaconMetadata) + sizeof(bss.ies)] = {};
00692:             RTW88WCLBeaconMetadata metadata = {};
00693:             metadata.ieLength = bss.ies_len;
00694:             metadata.channelSpec = static_cast<uint16_t>(bss.channel |
00695:                 (bss.channel <= 14 ? 0x1000 : 0xd000));
00696:             memcpy(metadata.ssid, bss.ssid, bss.ssid_len);
00697:             metadata.ssidLength = bss.ssid_len;
00698:             metadata.channel = bss.channel;
00699:             memcpy(metadata.bssid, bss.bssid, sizeof(metadata.bssid));
00700:             metadata.rssi = bss.rssi;
00701:             metadata.interval = bss.beacon_interval;
00702:             metadata.capability = static_cast<uint16_t>(bss.capabilities);
00703:             metadata.flags = kWCLBeaconRSSIValid;
00704:             if (bss.ssid_len) metadata.flags |= kWCLBeaconSSIDPresent;
00705:             memcpy(message, &metadata, sizeof(metadata));
00706:             if (bss.ies_len)
00707:                 memcpy(message + sizeof(metadata), bss.ies, bss.ies_len);
00708:             postWCLMessage(kWCLMessageScanResult, message,
00709:                            sizeof(metadata) + bss.ies_len);
00710:         }
00711:     }
00712: 
00713:     _scanActive = false;
00714:     _activeScanGeneration = 0;
00715:     setScanningState(_scanFilter.identifier, false, nullptr, 0);
00716:     postWCLMessage(kWCLMessageScanDone, &status, sizeof(status));
00717: }

00732: {
00733:     /* setWCL_LINK_UP_DONE may be delivered after a leave/abort request. Never
00734:      * publish an old backend CONNECTED state without a live join transaction. */

00750: void RTW88WCLInfraInterface::handleRTW88WCLRadioEvent(
00751:     const RTW88WCLRadioEvent &event)
00752: {
00753:     if (_stopping) return;
00754:     switch (event.event) {
00755:         case kRTW88EventScanDone: {
00756:             if (!_scanActive || event.scanGeneration != _activeScanGeneration)
00757:                 return; // stale completion from an aborted/superseded scan
00758:             const IOReturn status = event.value ? kIOReturnAborted
00759:                                                  : kIOReturnSuccess;
00760:             publishScanResults(status);
00761:             break;
00762:         }
00763:         case kRTW88EventConnected: {
00764:             if (!_activeJoinGeneration ||
00765:                 event.joinGeneration != _activeJoinGeneration)
00766:                 return;
00767:             if (!_assocNotified) {
00768:                 RTW88WCLAssociationEvent assoc = {};
00769:                 memcpy(assoc.bssid, _targetBSSID, sizeof(assoc.bssid));
00770:                 postWCLMessage(kWCLMessageAssocComplete, &assoc, sizeof(assoc));
00771:                 _assocNotified = true;
00772:             }
00773:             maybePublishConnected();
00774:             break;
00775:         }
00776:         case kRTW88EventAssocDone: {
00777:             if (!_activeJoinGeneration ||
00778:                 event.joinGeneration != _activeJoinGeneration)
00779:                 return; // stale association callback from an earlier join
00780:             if (!_assocNotified) {
00781:                 RTW88WCLAssociationEvent assoc = {};
00782:                 memcpy(assoc.bssid, _targetBSSID, sizeof(assoc.bssid));
00783:                 postWCLMessage(kWCLMessageAssocComplete, &assoc, sizeof(assoc));
00784:                 _assocNotified = true;
00785:             }
00786:             maybePublishConnected();
00787:             break;
00788:         }
00789:         case kRTW88EventDeauth: {
00790:             if (!_activeJoinGeneration ||
00791:                 event.joinGeneration != _activeJoinGeneration)
00792:                 return;
00793:             uint32_t reason = event.value;
00794:             postWCLMessage(kWCLMessageDeauth, &reason, sizeof(reason));
00795:             postConnectionComplete(1, static_cast<uint16_t>(reason),
00796:                                    kIOReturnNotAttached);
00797:             _connectNotified = false;
00798:             _assocNotified = false;
00799:             _activeJoinGeneration = 0;
00800:             setCurrentApAddress(nullptr);
00801:             setLinkState(kIO80211NetworkLinkDown, reason, false, 0);
00802:             break;
00803:         }
00804:         case kRTW88EventDisconnected: {
00805:             if (!_activeJoinGeneration ||
00806:                 event.joinGeneration != _activeJoinGeneration)
00807:                 return;
00808:             if (_connectNotified || _assocNotified)
00809:                 postConnectionComplete(1, 0, kIOReturnNotAttached);
00810:             _connectNotified = false;
00811:             _assocNotified = false;
00812:             _activeJoinGeneration = 0;
00813:             setCurrentApAddress(nullptr);
00814:             setLinkState(kIO80211NetworkLinkDown, 0, false, 0);
00815:             RTW88WCLLinkStatusEvent link = {};
00816:             memcpy(link.bssid, _targetBSSID, sizeof(link.bssid));
00817:             link.up = 0;
00818:             postWCLMessage(kWCLMessageLinkStatus, &link, sizeof(link));
00819:             break;
00820:         }
00821:         case kRTW88EventRSSIChanged: {
00822:             if (!_activeJoinGeneration ||
00823:                 event.joinGeneration != _activeJoinGeneration)
00824:                 return;
00825:             int rssi = 0;
00826:             if (_bridge.getRSSI(&rssi) == kIOReturnSuccess) {
00827:                 RTW88WCLLqmUpdate update = {};
00828:                 update.rssiValid = 1;
00829:                 update.rssi = rssi;
00830:                 postWCLMessage(kWCLMessageLqmUpdate, &update, sizeof(update));
00831:             }
00832:             break;
00833:         }
00834:     }
00835: }
